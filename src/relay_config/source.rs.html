1use std::collections::BTreeSet;
2use std::path::PathBuf;
3
4/// A custom [`serde_vars::source::FileSystem`], which loads files from the file system, but also
5/// keeps track of the files read.
6#[derive(Debug)]
7pub struct TrackingFileSystem<'a>(pub &'a mut BTreeSet<PathBuf>);
8
9impl<'a> serde_vars::source::FileSystem for TrackingFileSystem<'a> {
10    fn read(&mut self, path: &std::path::Path) -> std::io::Result<Vec<u8>> {
11        self.0.insert(path.to_owned());
12        std::fs::read(path)
13    }
14
15    fn read_to_string(&mut self, path: &std::path::Path) -> std::io::Result<String> {
16        self.0.insert(path.to_owned());
17        std::fs::read_to_string(path)
18    }
19}
