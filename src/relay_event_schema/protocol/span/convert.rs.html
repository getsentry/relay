1//! This module defines bidirectional field mappings between spans and transactions.
2
3use relay_conventions::attributes::{
4    BROWSER__NAME, HTTP__QUERY, SENTRY__ENVIRONMENT, SENTRY__EVENT__SERIALIZED_BREADCRUMBS,
5    SENTRY__EVENT__SERIALIZED_CONTEXTS, SENTRY__EVENT__SERIALIZED_EXTRA,
6    SENTRY__EVENT__SERIALIZED_META, SENTRY__EVENT__SERIALIZED_REQUEST, SENTRY__RELEASE,
7    SENTRY__SDK__NAME, SENTRY__SDK__VERSION, SENTRY__SEGMENT__NAME, URL__QUERY,
8};
9use relay_protocol::{Empty, IntoValue, MetaTree};
10
11use crate::protocol::{BrowserContext, Event, ProfileContext, Span, SpanData, TraceContext};
12
13impl From<&Event> for Span {
14    fn from(event: &Event) -> Self {
15        let Event {
16            transaction,
17
18            platform,
19            timestamp,
20            start_timestamp,
21            received,
22            release,
23            environment,
24            tags,
25            contexts,
26            breadcrumbs,
27            extra,
28            request,
29            measurements,
30            _metrics,
31            ..
32        } = event;
33
34        let trace = event.context::<TraceContext>();
35
36        // Fill data from trace context:
37        let mut data = trace
38            .map(|c| c.data.clone().map_value(SpanData::from))
39            .unwrap_or_default();
40
41        // Overwrite specific fields:
42        let span_data = data.get_or_insert_with(Default::default);
43        span_data.insert(
44            SENTRY__SEGMENT__NAME,
45            transaction.clone().map_value(IntoValue::into_value),
46        );
47        // For root spans, the name should just be the transaction name.
48        span_data.insert(
49            "sentry.name",
50            transaction.clone().map_value(IntoValue::into_value),
51        );
52        span_data.insert(
53            SENTRY__RELEASE,
54            release.clone().map_value(IntoValue::into_value),
55        );
56        span_data.insert(
57            SENTRY__ENVIRONMENT,
58            environment.clone().map_value(IntoValue::into_value),
59        );
60        if let Some(browser) = event.context::<BrowserContext>() {
61            span_data.insert(
62                BROWSER__NAME,
63                browser.name.clone().map_value(IntoValue::into_value),
64            );
65        }
66        if let Some(client_sdk) = event.client_sdk.value() {
67            span_data.insert(
68                SENTRY__SDK__NAME,
69                client_sdk.name.clone().map_value(IntoValue::into_value),
70            );
71            span_data.insert(
72                SENTRY__SDK__VERSION,
73                client_sdk.version.clone().map_value(IntoValue::into_value),
74            );
75        }
76        if let Some(request) = request.value()
77            && let Some(query) = request.query_string.value()
78            && let Some(qs) = query.to_query_string()
79        {
80            span_data.insert_value(HTTP__QUERY, format!("?{qs}"));
81            span_data.insert_value(URL__QUERY, qs);
82        }
83
84        if contexts.value().is_some_and(|c| !c.0.is_empty())
85            && let Ok(json) = contexts.payload_to_json()
86        {
87            span_data.insert_value(SENTRY__EVENT__SERIALIZED_CONTEXTS, json);
88        }
89        if breadcrumbs
90            .value()
91            .and_then(|b| b.values.value())
92            .is_some_and(|v| !v.is_empty())
93            && let Ok(json) = breadcrumbs.payload_to_json()
94        {
95            span_data.insert_value(SENTRY__EVENT__SERIALIZED_BREADCRUMBS, json);
96        }
97        if extra.value().is_some_and(|e| !e.is_empty())
98            && let Ok(json) = extra.payload_to_json()
99        {
100            span_data.insert_value(SENTRY__EVENT__SERIALIZED_EXTRA, json);
101        }
102        if request.value().is_some_and(|r| !r.is_empty())
103            && let Ok(json) = request.payload_to_json()
104        {
105            span_data.insert_value(SENTRY__EVENT__SERIALIZED_REQUEST, json);
106        }
107
108        let mut meta = MetaTree::default();
109        let contexts_meta = IntoValue::extract_meta_tree(contexts);
110        if !contexts_meta.is_empty() {
111            meta.children.insert("contexts".to_owned(), contexts_meta);
112        }
113        let breadcrumbs_meta = IntoValue::extract_meta_tree(breadcrumbs);
114        if !breadcrumbs_meta.is_empty() {
115            meta.children
116                .insert("breadcrumbs".to_owned(), breadcrumbs_meta);
117        }
118        let extra_meta = IntoValue::extract_meta_tree(extra);
119        if !extra_meta.is_empty() {
120            meta.children.insert("extra".to_owned(), extra_meta);
121        }
122        let request_meta = IntoValue::extract_meta_tree(request);
123        if !request_meta.is_empty() {
124            meta.children.insert("request".to_owned(), request_meta);
125        }
126        if !meta.is_empty()
127            && let Ok(json) = serde_json::to_string(&meta)
128        {
129            span_data.insert_value(SENTRY__EVENT__SERIALIZED_META, json);
130        }
131
132        Self {
133            timestamp: timestamp.clone(),
134            start_timestamp: start_timestamp.clone(),
135            exclusive_time: trace.map(|c| c.exclusive_time.clone()).unwrap_or_default(),
136            op: trace.map(|c| c.op.clone()).unwrap_or_default(),
137            span_id: trace.map(|c| c.span_id.clone()).unwrap_or_default(),
138            parent_span_id: trace.map(|c| c.parent_span_id.clone()).unwrap_or_default(),
139            trace_id: trace.map(|c| c.trace_id.clone()).unwrap_or_default(),
140            segment_id: trace.map(|c| c.span_id.clone()).unwrap_or_default(),
141            is_segment: true.into(),
142            // NB: Technically, this span may not be an actual remote span if this is a child
143            // transaction created within the same service as its parent. We still set `is_remote`
144            // as the best proxy to ensure this span will be detected as a segment by the spans
145            // pipeline.
146            is_remote: true.into(),
147            status: trace.map(|c| c.status.clone()).unwrap_or_default(),
148            description: transaction.clone(),
149            tags: tags.clone().map_value(|t| t.into()),
150            origin: trace.map(|c| c.origin.clone()).unwrap_or_default(),
151            profile_id: event
152                .context::<ProfileContext>()
153                .map(|c| c.profile_id.clone())
154                .unwrap_or_default(),
155            data,
156            links: trace.map(|c| c.links.clone()).unwrap_or_default(),
157            sentry_tags: Default::default(),
158            received: received.clone(),
159            measurements: measurements.clone(),
160            platform: platform.clone(),
161            was_transaction: true.into(),
162            kind: Default::default(),
163            other: Default::default(),
164        }
165    }
166}

334    #[test]
335    fn convert_preserves_contexts_breadcrumbs_extra() {
336        let event = Annotated::<Event>::from_json(
337            r#"{
338                "type": "transaction",
339                "transaction": "my transaction",
340                "contexts": {
341                    "browser": {"name": "Chrome"},
342                    "trace": {
343                        "trace_id": "4c79f60c11214eb38604f4ae0781bfb2",
344                        "span_id": "fa90fdead5f74052"
345                    }
346                },
347                "breadcrumbs": [
348                    {"type": "default", "category": "auth", "message": "login"}
349                ],
350                "extra": {
351                    "my_key": 1,
352                    "some_other_value": "foo bar"
353                }
354            }"#,
355        )
356        .unwrap()
357        .into_value()
358        .unwrap();
359
360        let span = Span::from(&event);
361        let data = span.data.value().unwrap();
362
363        assert_eq!(
364            data.get_str(SENTRY__EVENT__SERIALIZED_CONTEXTS),
365            Some(
366                r#"{"browser":{"name":"Chrome","type":"browser"},"trace":{"trace_id":"4c79f60c11214eb38604f4ae0781bfb2","span_id":"fa90fdead5f74052","type":"trace"}}"#
367            )
368        );
369        assert_eq!(
370            data.get_str(SENTRY__EVENT__SERIALIZED_BREADCRUMBS),
371            Some(r#"{"values":[{"type":"default","category":"auth","message":"login"}]}"#)
372        );
373        assert_eq!(
374            data.get_str(SENTRY__EVENT__SERIALIZED_EXTRA),
375            Some(r#"{"my_key":1,"some_other_value":"foo bar"}"#)
376        );
377    }

508    #[test]
509    fn convert_preserves_meta() {
510        let event = Annotated::<Event>::from_json(
511            r#"{
512                "type": "transaction",
513                "transaction": "my transaction",
514                "contexts": {
515                    "browser": {"name": "Chrome"},
516                    "trace": {
517                        "trace_id": "4c79f60c11214eb38604f4ae0781bfb2",
518                        "span_id": "fa90fdead5f74052"
519                    }
520                },
521                "breadcrumbs": [
522                    {"type": "default", "category": "auth", "message": "login"}
523                ],
524                "extra": {
525                    "my_key": "[Filtered]"
526                },
527                "request": {
528                    "data": {"note": "[Filtered]"}
529                },
530                "_meta": {
531                    "contexts": {
532                        "browser": {
533                            "name": {"": {"rem": [["browser_rule", "s"]]}}
534                        },
535                        "trace": {
536                            "trace_id": {"": {"rem": [["trace_rule", "s"]]}}
537                        }
538                    },
539                    "breadcrumbs": {
540                        "0": {
541                            "message": {"": {"rem": [["breadcrumb_rule", "s"]]}}
542                        }
543                    },
544                    "extra": {
545                        "my_key": {"": {"rem": [["extra_rule", "s", 0, 10]]}}
546                    },
547                    "request": {
548                        "data": {
549                            "note": {"": {"rem": [["request_rule", "s"]]}}
550                        }
551                    }
552                }
553            }"#,
554        )
555        .unwrap()
556        .into_value()
557        .unwrap();
558
559        let span = Span::from(&event);
560        let data = span.data.value().unwrap();
561
562        let meta = data.get_str(SENTRY__EVENT__SERIALIZED_META).unwrap();

564        assert_eq!(
