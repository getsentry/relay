1/// A pattern used to match allowed paths.
2///
3/// Scheme, domain and port are extracted from an url,
4/// they may be either a string (to be matched exactly, case insensitive)
5/// or None (matches anything in the respective position).
6#[derive(Hash, PartialEq, Eq)]
7pub struct SchemeDomainPort {
8    /// The scheme of the url.
9    pub scheme: Option<String>,
10    /// The domain of the url.
11    pub domain: Option<String>,
12    /// The port of the url.
13    pub port: Option<String>,
14}
15
16impl From<&str> for SchemeDomainPort {
17    /// parse a string into a SchemaDomainPort pattern
18    fn from(url: &str) -> SchemeDomainPort {
19        /// converts string into patterns for SchemeDomainPort
20        /// the convention is that a "*" matches everything which
21        /// we encode as a None (same as the absence of the pattern)
22        fn normalize(pattern: &str) -> Option<String> {
23            if pattern == "*" {
24                None
25            } else {
26                Some(pattern.to_lowercase())
27            }
28        }
29
30        //split the scheme from the url
31        let scheme_idx = url.find("://");
32        let (scheme, rest) = if let Some(idx) = scheme_idx {
33            (normalize(&url[..idx]), &url[idx + 3..]) // chop after the scheme + the "://" delimiter
34        } else {
35            (None, url) // no scheme, chop nothing form original string
36        };
37
38        // extract domain:port from the rest of the url
39        let end_domain_idx = rest.find('/');
40        let domain_port = if let Some(end_domain_idx) = end_domain_idx {
41            &rest[..end_domain_idx] // remove the path from rest
42        } else {
43            rest // no path, use everything
44        };
45
46        // split the domain and the port
47        let ipv6_end_bracket_idx = domain_port.rfind(']');
48        let port_separator_idx = if let Some(end_bracket_idx) = ipv6_end_bracket_idx {
49            // we have an ipv6 address, find the port separator after the closing bracket
50            domain_port[end_bracket_idx..]
51                .rfind(':')
52                .map(|x| x + end_bracket_idx)
53        } else {
54            // no ipv6 address, find the port separator in the whole string
55            domain_port.rfind(':')
56        };
57        let (domain, port) = if let Some(port_separator_idx) = port_separator_idx {
58            //we have a port separator, split the string into domain and port
59            (
60                normalize(&domain_port[..port_separator_idx]),
61                normalize(&domain_port[port_separator_idx + 1..]),
62            )
63        } else {
64            (normalize(domain_port), None) // no port, whole string represents the domain
65        };
66
67        SchemeDomainPort {
68            scheme,
69            domain,
70            port,
71        }
72    }
73}

75/// Checks if a url satisfies one of the specified origins.
76///
77/// An origin specification may be in any of the following formats:
78///
79///  - `http://domain.com[:port]`
80///  - an exact match is required
81///  - `*`: anything goes
82///  - `*.domain.com`: matches domain.com and any subdomains
83///  - `*:port`: matches any hostname as long as the port matches
84pub fn matches_any_origin(url: Option<&str>, origins: &[SchemeDomainPort]) -> bool {
85    // if we have a "*" (Any) option, anything matches so don't bother going forward
86    if origins
87        .iter()
88        .any(|o| o.scheme.is_none() && o.port.is_none() && o.domain.is_none())
89    {
90        return true;
91    }
92
93    if let Some(url) = url {
94        let url = SchemeDomainPort::from(url);
95
96        for origin in origins {
97            if origin.scheme.is_some() && url.scheme != origin.scheme {
98                continue; // scheme not matched
99            }
100            if origin.port.is_some() && url.port != origin.port {
101                continue; // port not matched
102            }
103            if origin.domain.is_some() && url.domain != origin.domain {
104                // no direct match for domain, look for  partial patterns (e.g. "*.domain.com")
105                if let (Some(origin_domain), Some(domain)) = (&origin.domain, &url.domain)
106                    && let Some(suffix) = origin_domain.strip_prefix('*')
107                    && host_matches_or_is_subdomain_of(
108                        domain,
109                        suffix.strip_prefix('.').unwrap_or(suffix),
110                    )
111                {
112                    return true; // partial domain pattern match
113                }
114                continue; // domain not matched
115            }
116            // if we are here all patterns have matched so we are done
117            return true;
118        }
119    }
120    false
121}

123/// Checks if `host` is `domain` itself or one of its subdomains.
124///
125/// The match only happens on a label boundary, so `notabc.com` and `bc.com` are not subdomains of
126/// `abc.com`. An empty `domain` never matches.
127pub fn host_matches_or_is_subdomain_of(host: &str, domain: &str) -> bool {
128    if domain.is_empty() {
129        return false;
130    }
131    host.strip_suffix(domain)
132        .is_some_and(|s| s.is_empty() || s.ends_with('.'))
133}

139    #[test]
140    fn test_host_matches_or_is_subdomain_of() {
141        for (host, domain, expected) in [
142            ("abc.com", "abc.com", true),
143            ("www.abc.com", "abc.com", true),
144            ("a.b.abc.com", "abc.com", true),
145            ("notabc.com", "abc.com", false),
146            ("bc.com", "abc.com", false),
147            ("abc.com.evil.org", "abc.com", false),
148            ("", "abc.com", false),
149            // an empty domain never matches
150            ("", "", false),
151            ("abc.com", "", false),
152            ("abc.", "", false),
153        ] {
154            assert_eq!(
155                host_matches_or_is_subdomain_of(host, domain),
156                expected,
157                "host `{host}` against domain `{domain}`"
158            );
159        }
160    }
