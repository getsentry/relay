3/// Returns `true` if the current process is the crash reporting process.
4pub fn is_crash_reporter_process() -> bool {
5    cfg!(feature = "crash-handler") && std::env::var_os(CRASH_REPORTER_PROCESS_ENV_VAR).is_some()
6}

8/// Builds the minidump integration for reporting fatal crashes.
9///
10/// The integration spawns a secondary process from the same binary during
11/// `sentry::init`. The two processes talk over IPC. If the primary process
12/// crashes, the secondary process captures a minidump and reports it to
13/// Sentry. The crashed process waits for the upload before it exits.
14///
15/// Returns `None` if no crash directory is configured.
16#[cfg(feature = "crash-handler")]
17pub fn integration(
18    config: &crate::SentryConfig,
19) -> Option<sentry::integrations::minidump::MinidumpIntegration> {
20    use sentry::integrations::minidump::MinidumpIntegration;
21
22    let db = config._crash_db.clone()?;
23
24    if !is_crash_reporter_process() {
25        crate::info!("Initializing crash handler in {}", db.display());
26    }
27
28    let integration = MinidumpIntegration::new()
29        .crashes_dir(db)
30        .server_env_var(CRASH_REPORTER_PROCESS_ENV_VAR)
31        .process_name("relay-crash")
32        .flush_timeout(std::time::Duration::from_secs(15))
33        .before_capture(|_scope, path| {
34            crate::info!("Minidump captured ({})", path.display());
35        });
36
37    // Make sure the implementations agree on the detection.
38    debug_assert_eq!(
39        is_crash_reporter_process(),
40        integration.is_crash_reporter_process()
41    );
42
43    Some(integration)
44}
