1use axum::extract::{DefaultBodyLimit, Query};
2use axum::response::IntoResponse;
3use axum::routing::{MethodRouter, post};
4use relay_config::ConfigSnapshot;
5use relay_event_schema::protocol::EventId;
6use serde::Deserialize;
7
8use crate::constants::UNREAL_USER_HEADER;
9use crate::endpoints::common::{self, BadStoreRequest, TextResponse};
10use crate::extractors::IntegrationBuilder;
11use crate::integrations::ErrorsIntegration;
12use crate::middlewares;
13use crate::service::ServiceState;
14
15#[derive(Debug, Deserialize)]
16struct UnrealQuery {
17    #[serde(rename = "UserID")]
18    user_id: Option<String>,
19}
20
21async fn handle(
22    state: ServiceState,
23    Query(query): Query<UnrealQuery>,
24    builder: IntegrationBuilder,
25) -> axum::response::Result<impl IntoResponse> {
26    if builder.is_empty() {
27        return Err(BadStoreRequest::EmptyBody.into());
28    }
29
30    let id = EventId::new();
31    let envelope = builder
32        .with_event_id(id)
33        .with_header_opt(UNREAL_USER_HEADER, query.user_id)
34        .with_type(ErrorsIntegration::Unreal)
35        .build();
36
37    // Never respond with a 429 since clients often retry these
38    common::handle_envelope(&state, envelope)
39        .await?
40        .ignore_rate_limits();
41
42    // The return here is only useful for consistency because the UE4 crash reporter doesn't
43    // care about it.
44    Ok(TextResponse(Some(id)))
45}
46
47pub fn route(config: &ConfigSnapshot) -> MethodRouter<ServiceState> {
48    post(handle)
49        .route_layer(DefaultBodyLimit::max(config.max_attachments_size()))
50        .route_layer(axum::middleware::from_fn(middlewares::content_length))
51}
