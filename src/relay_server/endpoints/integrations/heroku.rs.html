1//! Heroku Log Drain integration endpoint.
2//!
3//! This module handles HTTPS log drains from Heroku's Logplex system.
4//! Logplex sends batches of syslog-formatted messages via POST requests.
5//!
6//! Heroku cannot attach custom headers to drain requests, the DSN key must be passed in the
7//! `sentry_key` query parameter.
8
9use axum::extract::DefaultBodyLimit;
10use axum::http::StatusCode;
11use axum::response::IntoResponse;
12use axum::routing::{MethodRouter, post};
13use relay_config::ConfigSnapshot;
14use relay_dynamic_config::Feature;
15use relay_ourlogs::{
16    HEROKU_LOGPLEX_HEADER_DRAIN_TOKEN as DRAIN_TOKEN, HEROKU_LOGPLEX_HEADER_MSG_COUNT as MSG_COUNT,
17};
18
19use crate::endpoints::common;
20use crate::envelope::ContentType;
21use crate::extractors::{IntegrationBuilder, RawContentType};
22use crate::integrations::LogsIntegration;
23use crate::service::ServiceState;
24
25/// All routes configured for the Heroku integration.
26///
27/// The integration currently supports the following endpoints:
28///  - Heroku Log Drain (Logplex HTTPS drain)
29pub fn routes(config: &ConfigSnapshot) -> axum::Router<ServiceState> {
30    axum::Router::new()
31        .route("/logs", logs::route(config))
32        .route("/logs/", logs::route(config))
33}
34
35mod logs {
36    use super::*;
37
38    async fn handle(
39        content_type: RawContentType,
40        headers: http::HeaderMap,
41        state: ServiceState,
42        builder: IntegrationBuilder,
43    ) -> axum::response::Result<impl IntoResponse> {
44        if !matches!(
45            content_type.as_ref().parse::<ContentType>(),
46            Ok(ContentType::Logplex)
47        ) {
48            return Ok(StatusCode::UNSUPPORTED_MEDIA_TYPE);
49        }
50
51        let header = |name| headers.get(name).and_then(|h| h.to_str().ok());
52
53        let envelope = builder
54            .with_type(LogsIntegration::HerokuLogDrain)
55            .with_header_opt(MSG_COUNT, header(MSG_COUNT))
56            .with_header_opt(DRAIN_TOKEN, header(DRAIN_TOKEN))
57            .with_required_feature(Feature::HerokuLogDrainEndpoint)
58            .build();
59
60        common::handle_envelope(&state, envelope)
61            .await?
62            .check_rate_limits()?;
63
64        // Logplex expects an empty response body.
65        Ok(StatusCode::ACCEPTED)
66    }
67
68    pub fn route(config: &ConfigSnapshot) -> MethodRouter<ServiceState> {
69        post(handle).route_layer(DefaultBodyLimit::max(config.max_container_size()))
70    }
71}
