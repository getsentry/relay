51    async fn service_metrics(&self) {
52        for (service, metrics) in self.runtime.current_services_metrics().iter() {
53            metric!(
54                gauge(RelayGauges::ServiceUtilization) = metrics.utilization as u64,
55                service = service.name(),
56                instance_id = &service.instance_id().to_string(),
57            );
58        }
59    }

61    async fn tokio_metrics(&self) {
62        metric!(gauge(RuntimeGauges::NumIdleThreads) = self.rt_metrics.num_idle_threads() as u64);
63        metric!(gauge(RuntimeGauges::NumAliveTasks) = self.rt_metrics.num_alive_tasks() as u64);
64        metric!(
65            gauge(RuntimeGauges::BlockingQueueDepth) =
66                self.rt_metrics.blocking_queue_depth() as u64
67        );
68        metric!(
69            gauge(RuntimeGauges::NumBlockingThreads) =
70                self.rt_metrics.num_blocking_threads() as u64
71        );
72        metric!(
73            gauge(RuntimeGauges::NumIdleBlockingThreads) =
74                self.rt_metrics.num_idle_blocking_threads() as u64
75        );
76
77        metric!(
78            counter(RuntimeCounters::BudgetForcedYieldCount) +=
79                self.rt_metrics.budget_forced_yield_count()
80        );
81
82        metric!(gauge(RuntimeGauges::NumWorkers) = self.rt_metrics.num_workers() as u64);
83        for worker in 0..self.rt_metrics.num_workers() {
84            let worker_name = worker.to_string();
85
86            metric!(
87                gauge(RuntimeGauges::WorkerLocalQueueDepth) =
88                    self.rt_metrics.worker_local_queue_depth(worker) as u64,
89                worker = &worker_name,
90            );
91            metric!(
92                gauge(RuntimeGauges::WorkerMeanPollTime) =
93                    self.rt_metrics.worker_mean_poll_time(worker).as_secs_f64(),
94                worker = &worker_name,
95            );
96
97            metric!(
98                counter(RuntimeCounters::WorkerLocalScheduleCount) +=
99                    self.rt_metrics.worker_local_schedule_count(worker),
100                worker = &worker_name,
101            );
102            metric!(
103                counter(RuntimeCounters::WorkerNoopCount) +=
104                    self.rt_metrics.worker_noop_count(worker),
105                worker = &worker_name,
106            );
107            metric!(
108                counter(RuntimeCounters::WorkerOverflowCount) +=
109                    self.rt_metrics.worker_overflow_count(worker),
110                worker = &worker_name,
111            );
112            metric!(
113                counter(RuntimeCounters::WorkerParkCount) +=
114                    self.rt_metrics.worker_park_count(worker),
115                worker = &worker_name,
116            );
117            metric!(
118                counter(RuntimeCounters::WorkerPollCount) +=
119                    self.rt_metrics.worker_poll_count(worker),
120                worker = &worker_name,
121            );
122            metric!(
123                counter(RuntimeCounters::WorkerStealCount) +=
124                    self.rt_metrics.worker_steal_count(worker),
125                worker = &worker_name,
126            );
127            metric!(
128                counter(RuntimeCounters::WorkerStealOperations) +=
129                    self.rt_metrics.worker_steal_operations(worker),
130                worker = &worker_name,
131            );
132            metric!(
133                counter(RuntimeCounters::WorkerTotalBusyDuration) +=
134                    self.rt_metrics
135                        .worker_total_busy_duration(worker)
136                        .as_millis() as u64,
137                worker = &worker_name,
138            );
139        }
140    }

142    async fn upstream_status(&self) {
143        if self.config.current().relay_mode() == RelayMode::Managed
144            && let Ok(is_outage) = self.upstream_relay.send(IsNetworkOutage).await
145        {
146            metric!(gauge(RelayGauges::NetworkOutage) = u64::from(is_outage));
147        }
148    }

176    #[cfg(not(feature = "processing"))]
177    async fn redis_clients(&self) {}

179    #[cfg(feature = "processing")]
180    async fn redis_clients(&self) {
181        if let Some(RedisClients {
182            project_configs,
183            quotas,
184        }) = self.redis_clients.as_ref()
185        {
186            Self::async_redis_connection(project_configs, "project_configs");
187            Self::async_redis_connection(quotas, "quotas");
188        }
189    }
190
191    fn emit_async_pool_metrics<T>(async_pool: &AsyncPool<T>) {
192        let metrics = async_pool.metrics();
193
194        metric!(
195            gauge(RelayGauges::AsyncPoolQueueSize) = metrics.queue_size(),
196            pool = async_pool.name()
197        );
198        metric!(
199            gauge(RelayGauges::AsyncPoolUtilization) = metrics.cpu_utilization() as u64,
200            pool = async_pool.name()
201        );
202        metric!(
203            gauge(RelayGauges::AsyncPoolActivity) = metrics.activity() as u64,
204            pool = async_pool.name()
205        );
206        metric!(
207            counter(RelayCounters::AsyncPoolFinishedTasks) += metrics.finished_tasks(),
208            pool = async_pool.name()
209        );
210    }
211
212    async fn async_pools_metrics(&self) {
213        if let Some(processor_pool) = &self.processor_pool {
214            Self::emit_async_pool_metrics(processor_pool)
215        }
216        #[cfg(feature = "processing")]
217        Self::emit_async_pool_metrics(&self.store_pool);
218    }

221impl Service for RelayStats {
222    type Interface = ();
223
224    async fn run(self, mut rx: relay_system::Receiver<Self::Interface>) {
225        let Some(mut ticker) = self
226            .config
227            .current()
228            .metrics_periodic_interval()
229            .map(interval)
230        else {
231            return;
232        };
233
234        loop {
235            tokio::select! {
236                _ = async {
237                    tokio::join!(
238                        self.upstream_status(),
239                        self.service_metrics(),
240                        self.tokio_metrics(),
241                        self.redis_clients(),
242                        self.async_pools_metrics()
243                    );
244                    ticker.tick().await;
245                } => {},
246                None = rx.recv() => break,
247            }
248        }
249    }
250}

273    #[test]
274    fn test_shutdown_on_channel_close() {
275        let runtime = relay_system::Runtime::builder("stats-test")
276            .worker_threads(1)
277            .build();
278
279        runtime.block_on(async {
280            let (upstream, mut upstream_rx) = relay_system::channel("upstream");
281            let service = dummy_service(runtime.handle(), upstream);
282            let (addr, rx) = relay_system::channel(RelayStats::name());
283            let mut run = std::pin::pin!(service.run(rx));
284            assert!(futures::poll!(&mut run).is_pending());
285
286            // Keep the upstream response pending: shutdown must also interrupt a stats check.
287            let _request = upstream_rx.recv().await.unwrap();
288            drop(addr);
289            assert!(futures::poll!(&mut run).is_ready());
290        });
291    }
