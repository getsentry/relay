41/// Extracts the items from an Unreal 4 crash report payload.
42fn extract_items(payload: Bytes, config: &ConfigSnapshot) -> Result<Items, ProcessingError> {
43    let mut items = Items::new();
44    let crash = Unreal4Crash::parse_with_limit(&payload, config.max_envelope_size())?;
45
46    for file in crash.files().take(config.max_attachment_count()) {
47        let (content_type, attachment_type) = match file.ty() {
48            Unreal4FileType::Minidump => (ContentType::Minidump, AttachmentType::Minidump),
49            Unreal4FileType::AppleCrashReport => {
50                (ContentType::Text, AttachmentType::AppleCrashReport)
51            }
52            Unreal4FileType::Log => (ContentType::Text, AttachmentType::UnrealLogs),
53            Unreal4FileType::Config => (ContentType::OctetStream, AttachmentType::Attachment),
54            Unreal4FileType::Context => (ContentType::Xml, AttachmentType::UnrealContext),
55            _ => match file.name() {
56                self::ITEM_NAME_EVENT => (ContentType::MsgPack, AttachmentType::EventPayload),
57                self::ITEM_NAME_BREADCRUMBS1 => (ContentType::MsgPack, AttachmentType::Breadcrumbs),
58                self::ITEM_NAME_BREADCRUMBS2 => (ContentType::MsgPack, AttachmentType::Breadcrumbs),
59                name if name.ends_with(self::ATTACHMENT_NV_GPUDUMP_SUFFIX) => {
60                    (ContentType::OctetStream, AttachmentType::NvGpuDump)
61                }
62                name if name.ends_with(self::ATTACHMENT_NV_DBG_SUFFIX) => {
63                    (ContentType::OctetStream, AttachmentType::NvShaderDebug)
64                }
65                name if name.ends_with(self::ATTACHMENT_CSO_SUFFIX) => {
66                    (ContentType::OctetStream, AttachmentType::CompiledShader)
67                }
68                _ => (ContentType::OctetStream, AttachmentType::Attachment),
69            },
70        };
71
72        let mut item = Item::new(ItemType::Attachment);
73        item.set_filename(file.name());
74        item.set_payload(content_type, file.into_bytes());
75        item.set_attachment_type(attachment_type);
76        items.push(item);
77    }
78
79    Ok(items)
80}

82/// Expands an Unreal 4 crash report payload and returns the expanded items.
83pub fn expand_unreal(
84    payload: Bytes,
85    config: &ConfigSnapshot,
86) -> Result<UnrealExpansion, ProcessingError> {
87    let items = extract_items(payload, config)?;
88
89    let mut context = items
90        .iter()
91        .find(|&item| matches!(item.attachment_type(), Some(AttachmentType::UnrealContext)))
92        .map(|item| Unreal4Context::parse(&item.payload()))
93        .transpose()?;
94
95    let event = context.as_mut().and_then(take_event_item);
96
97    Ok(UnrealExpansion {
98        event,
99        context,
100        attachments: items,
101    })
102}

156/// Merge an unreal logs object into event breadcrumbs.
157fn merge_unreal_logs(
158    event: &mut Event,
159    logs: impl IntoIterator<Item = Bytes>,
160) -> Result<(), Unreal4Error> {
161    let breadcrumbs = event
162        .breadcrumbs
163        .value_mut()
164        .get_or_insert_with(Values::default)
165        .values
166        .value_mut()
167        .get_or_insert_with(Array::default);
168
169    for breadcrumb in logs
170        .into_iter()
171        .flat_map(parse_unreal_logs)
172        .take(MAX_NUM_UNREAL_LOGS)
173    {
174        breadcrumbs.push(Annotated::new(breadcrumb?));
175    }
176
177    Ok(())
178}

180/// Parses unreal logs into breadcrumbs.
181fn parse_unreal_logs(data: Bytes) -> impl Iterator<Item = Result<Breadcrumb, Unreal4Error>> {
182    let (logs, err) = match Unreal4LogEntry::parse(&data, MAX_NUM_UNREAL_LOGS) {
183        Ok(logs) => (logs, None),
184        Err(err) => (Vec::new(), Some(Err(err))),
185    };
186
187    let logs = logs.into_iter().map(|log| {
188        let timestamp = log
189            .timestamp
190            .and_then(|ts| {
191                Utc.timestamp_opt(ts.unix_timestamp(), ts.nanosecond())
192                    .latest()
193            })
194            .map(Timestamp);
195
196        Ok(Breadcrumb {
197            timestamp: Annotated::from(timestamp),
198            category: Annotated::from(log.component),
199            message: Annotated::new(log.message),
200            ..Breadcrumb::default()
201        })
202    });
203
204    err.into_iter().chain(logs)
205}

231/// Merges an unreal context object into an event
232fn merge_unreal_context(event: &mut Event, context: Unreal4Context) {
233    let mut runtime_props = match context.runtime_properties {
234        Some(runtime_props) => runtime_props,
235        None => return,
236    };
237
238    if let Some(msg) = runtime_props.error_message.take() {
239        let logentry = event.logentry.get_or_insert_with(LogEntry::default);
240        if logentry.formatted.is_empty() {
241            logentry.formatted = Annotated::new(Message::from(msg));
242        }
243    }
244
245    if let Some(username) = runtime_props.username.take() {
246        let user = event.user.get_or_insert_with(User::default);
247        if user.username.is_empty() {
248            user.username.set_value(Some(username.into()));
249        }
250    }
251
252    if let Some(login_id) = &runtime_props.login_id {
253        let id = event.user.get_or_insert_with(User::default).id.value_mut();
254
255        if id.as_ref().is_none_or(|s| s.is_empty()) {
256            *id = Some(login_id.clone().into());
257        }
258    }
259
260    let contexts = event.contexts.get_or_insert_with(Contexts::default);
261
262    if let Some(memory_physical) = runtime_props.memory_stats_total_physical.take() {
263        let device_context = contexts.get_or_default::<DeviceContext>();
264        if device_context.memory_size.is_empty() {
265            device_context.memory_size = Annotated::new(memory_physical);
266        }
267    }
268
269    // OS information is likely overwritten by Minidump processing later.
270    if let Some(os_major) = runtime_props.misc_os_version_major.take() {
271        let os_context = contexts.get_or_default::<OsContext>();
272        if os_context.raw_description.is_empty() {
273            os_context.raw_description = Annotated::new(os_major);
274        }
275    }
276
277    // See https://github.com/EpicGames/UnrealEngine/blob/5.3.2-release/Engine/Source/Runtime/RHI/Private/DynamicRHI.cpp#L368-L376
278    if let Some(adapter_name) = context.engine_data.get("RHI.AdapterName") {
279        let gpu_context = contexts.get_or_default::<GpuContext>();
280        if gpu_context.name.is_empty() {
281            gpu_context.name = Annotated::new(adapter_name.into());
282        }
283    } else if let Some(gpu_brand) = runtime_props.misc_primary_gpu_brand.take() {
284        let gpu_context = contexts.get_or_default::<GpuContext>();
285        if gpu_context.name.is_empty() {
286            gpu_context.name = Annotated::new(gpu_brand);
287        }
288    }
289
290    if let Some(device_id) = context.engine_data.get("RHI.DeviceId") {
291        let gpu_context = contexts.get_or_default::<GpuContext>();
292        if gpu_context.id.is_empty() {
293            gpu_context.id = Annotated::new(Value::String(device_id.into()));
294        }
295    }
296
297    if let Some(feature_level) = context.engine_data.get("RHI.FeatureLevel") {
298        let gpu_context = contexts.get_or_default::<GpuContext>();
299        if gpu_context.graphics_shader_level.is_empty() {
300            gpu_context.graphics_shader_level = Annotated::new(feature_level.into());
301        }
302    }
303
304    if let Some(vendor_name) = context.engine_data.get("RHI.GPUVendor") {
305        let gpu_context = contexts.get_or_default::<GpuContext>();
306        if gpu_context.vendor_name.is_empty() {
307            gpu_context.vendor_name = Annotated::new(vendor_name.into());
308        }
309    }
310
311    if let Some(driver_version) = context.engine_data.get("RHI.UserDriverVersion") {
312        let gpu_context = contexts.get_or_default::<GpuContext>();
313        if gpu_context.version.is_empty() {
314            gpu_context.version = Annotated::new(driver_version.into());
315        }
316    }
317
318    if let Some(rhi_name) = context.engine_data.get("RHI.RHIName") {
319        let gpu_context = contexts.get_or_default::<GpuContext>();
320        if gpu_context.api_type.is_empty() {
321            gpu_context.api_type = Annotated::new(rhi_name.into());
322        }
323    }
324
325    if event.level.is_empty() {
326        if runtime_props.is_assert.unwrap_or(false) {
327            event.level = Annotated::new(Level::Error)
328        } else if runtime_props.is_ensure.unwrap_or(false) {
329            event.level = Annotated::new(Level::Warning)
330        }
331    }
332
333    // Modules are not used and later replaced with Modules from the Minidump or Apple Crash Report.
334    runtime_props.modules.take();
335
336    // Promote all game data (except the special `__sentry` key) into a context.
337    if !context.game_data.is_empty() {
338        let game_context = contexts.get_or_insert_with("game", || Context::Other(Object::new()));
339        if let Context::Other(game_context) = game_context {
340            let filtered_keys = context
341                .game_data
342                .into_iter()
343                .filter(|(key, _)| key != SENTRY_CRASH_PAYLOAD_KEY)
344                .map(|(key, value)| (key, Annotated::new(Value::String(value))));
345
346            game_context.extend(filtered_keys);
347        }
348    }
349
350    // Add sdk information for analytics.
351    event.client_sdk.get_or_insert_with(|| ClientSdkInfo {
352        name: CLIENT_SDK_NAME.to_owned().into(),
353        version: runtime_props
354            .crash_reporter_client_version
355            .take()
356            .unwrap_or_else(|| "0.0.0".to_owned())
357            .into(),
358        ..ClientSdkInfo::default()
359    });
360
361    if let Ok(Some(Value::Object(props))) = relay_protocol::to_value(&runtime_props) {
362        let unreal_context =
363            contexts.get_or_insert_with("unreal", || Context::Other(Object::new()));
364
365        if let Context::Other(unreal_context) = unreal_context {
366            unreal_context.extend(props);
367        }
368    }
369}

380/// Parses one line of the legacy `CallStack` into a frame.
381///
382/// Lines look like `Module!Function(...) [File:Line]`, where everything after the module is
383/// optional. Unsymbolicated frames are written as `Module!UnknownFunction(0xOffset)`, and symbols
384/// without debug information carry their offset, like `Module!Function(+0x8a)`. The module can
385/// also be omitted, like in `UnknownFunction []`.
386///
387/// Returns the frame and the module offset of an unsymbolicated frame.
388fn parse_legacy_frame(line: &str) -> Option<(Frame, Option<u64>)> {
389    let line = line.trim();
390    if line.is_empty() {
391        return None;
392    }
393
394    let pairs = CallStackParser::parse(Rule::legacy_frame, line).ok()?;
395
396    let mut frame = Frame::default();
397    let mut function = "";
398    let mut offset = None;
399    let mut has_offset = false;
400
401    for pair in pairs.flatten() {
402        match pair.as_rule() {
403            Rule::module | Rule::module_only if !pair.as_str().is_empty() => {
404                frame.package = Annotated::new(pair.as_str().to_owned());
405            }
406            Rule::function => function = pair.as_str(),
407            Rule::module_offset | Rule::export_offset => has_offset = true,
408            Rule::offset => offset = u64::from_str_radix(pair.as_str(), 16).ok(),
409            Rule::file => frame.filename = Annotated::new(pair.as_str().into()),
410            Rule::lineno => frame.lineno = Annotated::from(pair.as_str().parse::<u64>().ok()),
411            _ => {}
412        }
413    }
414
415    // Symbols with an offset are the closest exported symbol, found without debug information.
416    // They are often wrong and would group unrelated crashes together, so they are skipped. The
417    // last line is cut off when the call stack exceeds the engine's buffer.
418    let is_cut_off = !(function.ends_with(')') && function.contains('('))
419        && function.matches('(').count() != function.matches(')').count();
420    let is_symbol =
421        !has_offset && !is_cut_off && !function.is_empty() && function != "UnknownFunction";
422
423    if is_symbol {
424        frame.function = Annotated::new(function.to_owned());
425    }
426
427    Some((frame, offset))
428}

430/// A frame of the portable `PCallStack`.
431#[derive(Debug, PartialEq)]
432struct PortableFrame<'a> {
433    module: Option<&'a str>,
434    offset: u64,
435    address: u64,
436}
437
438/// Parses the portable `PCallStack`.
439///
440/// Returns `None` if the call stack cannot be parsed up to the frame limit or contains
441/// placeholder addresses.
442fn parse_portable_call_stack(call_stack: &str) -> Option<Vec<PortableFrame<'_>>> {
443    let pairs = CallStackParser::parse(Rule::portable_call_stack, call_stack).ok()?;
444
445    let mut frames = Vec::new();
446    for frame in pairs.flatten() {
447        if frames.len() >= MAX_NUM_UNREAL_FRAMES {
448            break;
449        }
450        match frame.as_rule() {
451            Rule::portable_frame => {}
452            // Addresses are paired with the `CallStack` frames by position, so a call stack that
453            // does not parse to the end is not used at all; the `CallStack` frames are then kept
454            // without addresses. This is not reached past the frame limit, where the rest is
455            // ignored.
456            Rule::truncated => return None,
457            _ => continue,
458        }
459
460        let mut module = None;
461        let mut base = None;
462        let mut offset = None;
463        for pair in frame.into_inner() {
464            match pair.as_rule() {
465                Rule::portable_module => module = Some(pair.as_str()),
466                Rule::image_base => base = u64::from_str_radix(pair.as_str(), 16).ok(),
467                Rule::portable_offset => offset = u64::from_str_radix(pair.as_str(), 16).ok(),
468                _ => {}
469            }
470        }
471
472        let (base, offset) = (base?, offset?);
473        // The frame has no address, so the call stack cannot be paired by position.
474        if base == UNKNOWN_IMAGE_BASE {
475            return None;
476        }
477
478        frames.push(PortableFrame {
479            module,
480            offset,
481            address: base.checked_add(offset)?,
482        });
483    }
484
485    Some(frames)
486}

488/// Returns whether a module of the legacy call stack is the module of the portable call stack.
489///
490/// The legacy call stack can differ in case, replace `-` with `_`, or add a version suffix, like
491/// `libc.so.6` for `libc.so`.
492fn is_same_module(legacy: &str, portable: &str) -> bool {
493    fn normalize(c: u8) -> u8 {
494        match c {
495            b'-' => b'_',
496            c => c.to_ascii_lowercase(),
497        }
498    }
499
500    let (legacy, portable) = (legacy.as_bytes(), portable.as_bytes());
501    let len = legacy.len().min(portable.len());
502
503    let same_prefix = legacy[..len]
504        .iter()
505        .zip(&portable[..len])
506        .all(|(a, b)| normalize(*a) == normalize(*b));
507
508    // The longer module may only continue with a version suffix, like `.6`.
509    same_prefix
510        && legacy
511            .get(len)
512            .or_else(|| portable.get(len))
513            .is_none_or(|c| *c == b'.')
514}

516/// Returns whether the engine writes full image bases into the portable call stack.
517///
518/// Unreal Engine 4.25 and older truncate image bases to 32 bits.
519fn has_full_image_bases(engine_version: Option<&str>) -> bool {
520    let mut parts = engine_version.unwrap_or_default().split(['.', '-']);
521    let major = parts.next().and_then(|part| part.parse::<u32>().ok());
522    let minor = parts.next().and_then(|part| part.parse::<u32>().ok());
523
524    match (major, minor) {
525        (Some(major), Some(minor)) => (major, minor) >= (4, 26),
526        _ => true,
527    }
528}

530/// Creates an exception with the stack trace from the Unreal crash context.
531///
532/// Platforms that do not write a minidump, like Linux, still store the crashing call stack in the
533/// crash context. Frames are taken from the legacy `CallStack`, and their instruction addresses
534/// from the portable `PCallStack` if it has at least as many frames and both agree on every frame.
535fn unreal_exception(runtime_props: &Unreal4ContextRuntimeProperties) -> Option<Exception> {
536    let legacy: Vec<(Frame, Option<u64>)> = runtime_props
537        .legacy_call_stack
538        .as_deref()
539        .unwrap_or_default()
540        .lines()
541        .filter_map(parse_legacy_frame)
542        .take(MAX_NUM_UNREAL_FRAMES)
543        .collect();
544
545    let portable = runtime_props
546        .portable_call_stack
547        .as_deref()
548        .filter(|_| has_full_image_bases(runtime_props.engine_version.as_deref()))
549        .and_then(parse_portable_call_stack)
550        .unwrap_or_default();
551
552    // Unknown functions are written with the offset of their return address minus one, to point
553    // into the call instruction. The engine does not do this for the crashing frame, which is the
554    // first frame unless the frames of the crash handler were skipped.
555    let stacks_match = legacy.len() <= portable.len()
556        && legacy
557            .iter()
558            .zip(&portable)
559            .enumerate()
560            .all(|(index, ((frame, offset), portable))| {
561                let same_module = match (frame.package.as_str(), portable.module) {
562                    (Some(legacy), Some(portable)) => is_same_module(legacy, portable),
563                    // Without a module, only the offset can confirm the frame. Frames without
564                    // an offset are only kept in place if they have nothing to misattribute.
565                    (None, _) => {
566                        offset.is_some() || (frame.function.is_empty() && frame.filename.is_empty())
567                    }
568                    (Some(_), None) => false,
569                };
570                let same_offset = offset.is_none_or(|offset| {
571                    offset.checked_add(1) == Some(portable.offset)
572                        || (index == 0 && offset == portable.offset)
573                });
574                same_module && same_offset
575            });
576
577    let mut frames: Vec<Frame> = legacy.into_iter().map(|(frame, _)| frame).collect();
578
579    if stacks_match {
580        for (frame, portable) in frames.iter_mut().zip(&portable) {
581            frame.instruction_addr = Annotated::new(Addr(portable.address));
582            if frame.package.is_empty() {
583                frame.package = Annotated::from(portable.module.map(str::to_owned));
584            }
585        }
586
587        // The legacy call stack can be shorter, for example it may omit system libraries.
588        let missing = portable.get(frames.len()..).unwrap_or_default();
589        frames.extend(missing.iter().map(|portable| Frame {
590            package: Annotated::from(portable.module.map(str::to_owned)),
591            instruction_addr: Annotated::new(Addr(portable.address)),
592            ..Frame::default()
593        }));
594    }
595
596    if frames.is_empty() {
597        return None;
598    }
599
600    // Unreal lists the crashing frame first, Sentry expects it last.
601    frames.reverse();
602
603    let crash_type = runtime_props
604        .crash_type
605        .as_deref()
606        .filter(|crash_type| !crash_type.is_empty());
607    let is_ensure = runtime_props.is_ensure.unwrap_or(false) || crash_type == Some("Ensure");
608    let is_stall = runtime_props
609        .custom
610        .get("IsStall")
611        .is_some_and(|v| v == "true")
612        || crash_type == Some("Stall");
613
614    let ty = match crash_type {
615        Some(crash_type) => crash_type,
616        None if is_ensure => "Ensure",
617        None if is_stall => "Stall",
618        None if runtime_props.is_assert.unwrap_or(false) => "Assert",
619        None => "Crash",
620    };
621
622    // These reports are sent while the application keeps running.
623    let handled = is_ensure || is_stall || crash_type == Some("VerseRuntimeError");
624
625    Some(Exception {
626        ty: Annotated::new(ty.to_owned()),
627        value: Annotated::from(runtime_props.error_message.clone().map(JsonLenientString)),
628        stacktrace: Annotated::new(Stacktrace(RawStacktrace {
629            frames: Annotated::new(frames.into_iter().map(Annotated::new).collect()),
630            ..RawStacktrace::default()
631        })),
632        mechanism: Annotated::new(Mechanism {
633            ty: Annotated::new("unreal".to_owned()),
634            handled: Annotated::new(handled),
635            synthetic: Annotated::new(true),
636            ..Mechanism::default()
637        }),
638        ..Exception::default()
639    })
640}

643fn has_debug_images(event: &Event) -> bool {
644    event
645        .debug_meta
646        .value()
647        .and_then(|debug_meta| debug_meta.images.value())
648        .is_some_and(|images| !images.is_empty())

651/// Returns whether a frame of the exception has a function or file name.
652fn has_symbols(exception: &Exception) -> bool {
653    let frames = exception
654        .stacktrace
655        .value()
656        .and_then(|stacktrace| stacktrace.frames.value());
657
658    frames.into_iter().flatten().any(|frame| {
659        frame
660            .value()
661            .is_some_and(|frame| !frame.function.is_empty() || !frame.filename.is_empty())
662    })

665/// Processes an unreal crash report.
666///
667/// The `user_header` should be extracted from the [`crate::constants::UNREAL_USER_HEADER`] envelope header.
668///
669/// With `context_stacktrace`, crashes without a minidump or Apple crash report get an exception
670/// with the stack trace from the crash context, if its frames have names or the event has debug
671/// images.
672pub fn process_unreal(
673    context: Option<Unreal4Context>,
674    event_id: EventId,
675    event: &mut Annotated<Event>,
676    attachments: &[Item],
677    user_header: Option<&str>,
678    context_stacktrace: bool,
679) -> Result<Option<ProcessedUnrealReport>, Unreal4Error> {
680    let mut logs_items = attachments
681        .iter()
682        .filter(|item| item.attachment_type() == Some(AttachmentType::UnrealLogs))
683        .map(|item| item.payload())
684        .peekable();
685
686    // Early exit if there is no information.
687    if context.is_none() && user_header.is_none() && logs_items.peek().is_none() {
688        return Ok(None);
689    }
690
691    // If we have UE4 info, ensure an event is there to fill. DO NOT fill if there is no unreal
692    // information, or otherwise envelope processing breaks.
693    let event = event.get_or_insert_with(Event::default);
694
695    if let Some(user_info) = user_header {
696        merge_unreal_user_info(event, user_info);
697    }
698
699    merge_unreal_logs(event, logs_items)?;
700
701    let mut user_reports = Items::new();
702    if let Some(mut context) = context {
703        if let Some(report) = get_unreal_user_report(event_id, &mut context) {
704            user_reports.push(report);
705        }
706
707        // Minidumps and Apple crash reports are processed later and provide a better stack trace.
708        let has_native_crash = attachments.iter().any(|item| {
709            matches!(
710                item.attachment_type(),
711                Some(AttachmentType::Minidump | AttachmentType::AppleCrashReport)
712            )
713        });
714
715        if context_stacktrace
716            && !has_native_crash
717            && event.exceptions.is_empty()
718            && let Some(exception) = context
719                .runtime_properties
720                .as_ref()
721                .and_then(unreal_exception)
722                // Frames without names can only be symbolicated with debug images. Without
723                // either, the stack trace adds nothing but would still change grouping.
724                .filter(|exception| has_debug_images(event) || has_symbols(exception))
725        {
726            event.exceptions = Annotated::new(Values::new(vec![Annotated::new(exception)]));
727            // The frames are native, like those of a minidump.
728            *event.platform.value_mut() = Some("native".to_owned());
729        }
730
731        merge_unreal_context(event, context);
732    }
733
734    // This matches the native processing, minidumps and other crash reports which do not have a
735    // dedicated level set are assumed to be fatal.
736    //
737    // We update this at the very end where we can be sure all event data has already been merged into
738    // the event (e.g. the unreal context which may set the level).
739    event.level.get_or_insert_with(|| Level::Fatal);
740
741    Ok(Some(ProcessedUnrealReport { user_reports }))
742}

747    pub user_reports: Items,
748}
