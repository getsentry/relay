13/// Decompresses request bodies and records the ratio of bytes read after and before decompression.
14///
15/// Use this with [`axum::middleware::from_fn`].
16pub async fn decompress(mut request: Request, next: Next) -> Response {
17    let matched_path = request.extract_parts::<MatchedPath>().await;
18    let route = matched_path.as_ref().map_or("unknown", |m| m.as_str());
19    let content_encoding = metrics::content_encoding_tag(&request);
20    let metrics = DecompressionMetrics::new(route.to_owned().into(), content_encoding);
21    let request = request.map(|body| metrics.count_encoded(body));
22    let decompression =
23        RequestDecompression::new(next.map_request(|request: Request<DecompressionBody<_>>| {
24            request.map(|body| Body::new(metrics.count_decoded(body)))
25        }));
26    let Ok(response) = decompression.oneshot(request).await;
27    response.map(Body::new)
28}

30/// Map request middleware that removes empty content encoding headers.
31///
32/// This is to be used along with [`decompress`].
33pub fn remove_empty_encoding(mut request: Request) -> Request {
34    if let header::Entry::Occupied(entry) = request.headers_mut().entry(header::CONTENT_ENCODING)
35        && should_ignore_encoding(entry.get().as_bytes())
36    {
37        entry.remove();
38    }
39
40    request
41}

43/// Returns `true` if this content-encoding value should be ignored.
44fn should_ignore_encoding(value: &[u8]) -> bool {
45    // sentry-ruby/5.x sends an empty string
46    // sentry.java.android/2.0.0 sends "UTF-8"
47    value == b"" || value.eq_ignore_ascii_case(b"utf-8")
48}
