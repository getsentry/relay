10impl AttachmentItem {
11    /// Returns the effective attachment type, defaulting to [`AttachmentType::Attachment`].
12    ///
13    /// Missing, invalid, and unknown headers use the default unless the payload is recognized
14    /// as a Nintendo Switch dying message. The original header is preserved for forwarding
15    /// and [`super::Item::is_unknown`].
16    pub fn attachment_type(&self) -> AttachmentType {
17        if let Some(ty) = self.data().headers().get(ItemHeaderKey::AttachmentType) {
18            return ty;
19        }
20
21        // Unfortunately when the switch protocol was decided on, it was missed to assign it a new
22        // attachment type, that's why we have to infer it here from the filename and contents.
23        if self.filename() == Some(crate::constants::NNSWITCH_DYING_MESSAGE_FILENAME)
24            && self
25                .data()
26                .payload_ref()
27                .starts_with(crate::constants::NNSWITCH_SENTRY_MAGIC)
28        {
29            return AttachmentType::NintendoSwitchDyingMessage;
30        }
31
32        AttachmentType::Attachment
33    }
34
35    /// Sets the attachment type of this item.
36    pub fn set_attachment_type(&mut self, attachment_type: AttachmentType) {
37        self.data_mut()
38            .headers_mut()
39            .set(ItemHeaderKey::AttachmentType, attachment_type);
40    }
41
42    /// Returns the attachment file name.
43    pub fn filename(&self) -> Option<&str> {
44        self.data().headers().get(ItemHeaderKey::Filename)
45    }
46
47    /// Sets the file name of this item.
48    pub fn set_filename<S>(&mut self, filename: S)
49    where
50        S: Into<String>,
51    {
52        self.data_mut()
53            .headers_mut()
54            .set(ItemHeaderKey::Filename, filename.into());
55    }
56
57    /// Returns the objectstore key for a stored attachment.
58    pub fn stored_key(&self) -> Option<&str> {
59        self.data().headers().stored_key()
60    }
61
62    /// Sets the objectstore key of this attachment item.
63    pub fn set_stored_key(&mut self, stored_key: String) {
64        self.data_mut().headers_mut().set_stored_key(stored_key);
65    }
66
67    /// Returns the length of the optional attachment metadata segment.
68    pub fn meta_length(&self) -> Option<u32> {
69        self.data().headers().get(ItemHeaderKey::MetaLength)
70    }
71
72    /// Sets the length of the optional meta segment.
73    pub fn set_meta_length(&mut self, meta_length: u32) {
74        self.data_mut()
75            .headers_mut()
76            .set(ItemHeaderKey::MetaLength, meta_length);
77    }
78
79    /// Sets the original payload length for an attachment with [`ContentType::AttachmentRef`].
80    pub fn set_attachment_length(&mut self, original_length: usize) {
81        debug_assert!(self.is_attachment_ref());
82        self.data_mut()
83            .headers_mut()
84            .set(ItemHeaderKey::AttachmentLength, original_length);
85    }
86
87    /// Returns the parent entity associated with this attachment, if any.
88    pub fn parent_id(&self) -> Option<ParentId> {
89        // Currently only `span_id` supported in `ParentId`.
90        let span_id = self.data().headers().get(ItemHeaderKey::SpanId)?;
91        Some(ParentId::SpanId(span_id))
92    }
93
94    /// Sets the parent entity that this item is associated with.
95    pub fn set_parent_id(&mut self, parent_id: Option<ParentId>) {
96        let span_id = parent_id.map(|p| match p {
97            ParentId::SpanId(span_id) => span_id,
98        });
99        self.data_mut()
100            .headers_mut()
101            .set_or_remove(ItemHeaderKey::SpanId, span_id);
102    }
103
104    /// Returns `true` if this item is an attachment with AttachmentV2 content type.
105    fn is_attachment_v2(&self) -> bool {
106        self.data().content_type() == Some(ContentType::TraceAttachment)
107    }
108
109    /// Returns `true` if this item is a V2 attachment owned by spans.
110    pub fn is_span_attachment(&self) -> bool {
111        self.is_attachment_v2() && matches!(self.parent_id(), Some(ParentId::SpanId(_)))
112    }
113
114    /// Returns `true` if this item is a V2 attachment without any span/log/etc. association.
115    pub fn is_trace_attachment(&self) -> bool {
116        self.is_attachment_v2() && self.parent_id().is_none()
117    }
118
119    /// Returns `true` if this item is an attachment placeholder.
120    pub fn is_attachment_ref(&self) -> bool {
121        self.data().content_type() == Some(ContentType::AttachmentRef)
122    }
123
124    /// Returns the [`AttachmentParentType`] of an attachment.
125    ///
126    /// For standard attachments (V1) always returns [`AttachmentParentType::Event`].
127    pub fn attachment_parent_type(&self) -> AttachmentParentType {
128        let is_trace_attachment = self.data().content_type() == Some(ContentType::TraceAttachment);
129
130        if is_trace_attachment {
131            match self.parent_id() {
132                Some(ParentId::SpanId(_)) => AttachmentParentType::Span,
133                None => AttachmentParentType::Trace,
134            }
135        } else {
136            AttachmentParentType::Event
137        }
138    }
139
140    /// Returns the attachment payload size.
141    ///
142    /// - For trace attachments, returns only the size of the actual payload, excluding the attachment meta.
143    /// - For attachment placeholders, returns the size represented by the placeholder.
144    /// - For Attachment, returns the size of entire payload.
145    ///
146    /// **Note:** This relies on the `meta_length` header which might not be correct as such this
147    /// is best effort.
148    pub fn attachment_body_size(&self) -> usize {
149        if self.is_attachment_v2() {
150            self.data()
151                .len()
152                .saturating_sub(self.meta_length().unwrap_or(0) as usize)
153        } else if self.is_attachment_ref() {
154            self.data()
155                .headers()
156                .get(ItemHeaderKey::AttachmentLength)
157                .unwrap_or(0)
158        } else {
159            self.data().len()
160        }
161        .max(1)
162    }
163
164    /// Whether this attachment contains data used to build an event.
165    ///
166    /// Span and trace attachments never create events, regardless of subtype.
167    pub fn carries_event_payload(&self) -> bool {
168        if !matches!(self.attachment_parent_type(), AttachmentParentType::Event) {
169            return false;
170        }
171        match self.attachment_type() {
172            AttachmentType::Minidump
173            | AttachmentType::AppleCrashReport
174            | AttachmentType::EventPayload
175            | AttachmentType::Prosperodump
176            | AttachmentType::Breadcrumbs
177            | AttachmentType::NvGpuDump
178            | AttachmentType::NintendoSwitchDyingMessage => true,
179            AttachmentType::Attachment
180            | AttachmentType::UnrealContext
181            | AttachmentType::UnrealLogs
182            | AttachmentType::ViewHierarchy
183            | AttachmentType::Flamegraph
184            | AttachmentType::NvShaderDebug
185            | AttachmentType::CompiledShader => false,
186        }
187    }
188}

190impl Counted for AttachmentItem {
191    fn quantities(&self) -> Quantities {
192        if self.data().rate_limited() {
193            return Quantities::new();
194        }
195
196        // New attachment types require an explicit accounting decision.
197        match self.attachment_type() {
198            AttachmentType::NintendoSwitchDyingMessage => {
199                // Event-parent dying messages are merged into event data.
200                if matches!(self.attachment_parent_type(), AttachmentParentType::Event) {
201                    return Quantities::new();
202                }
203            }
204            AttachmentType::Attachment
205            | AttachmentType::Minidump
206            | AttachmentType::AppleCrashReport
207            | AttachmentType::EventPayload
208            | AttachmentType::Breadcrumbs
209            | AttachmentType::Prosperodump
210            | AttachmentType::UnrealContext
211            | AttachmentType::UnrealLogs
212            | AttachmentType::ViewHierarchy
213            | AttachmentType::Flamegraph
214            | AttachmentType::NvGpuDump
215            | AttachmentType::NvShaderDebug
216            | AttachmentType::CompiledShader => {}
217        }
218
219        smallvec![
220            (DataCategory::Attachment, self.attachment_body_size()),
221            (DataCategory::AttachmentItem, 1),
222        ]
223    }
224}

228/// Parent identifier for an attachment-v2.
229///
230/// Attachments can be associated with different types of parent entities (only spans for now).
231///
232/// SpanId(None) indicates that the item is a span-attachment that is associated with no specific
233/// span.
234#[derive(Clone, Debug)]
235pub enum ParentId {
236    SpanId(Option<SpanId>),
237}
238
239impl ParentId {
240    /// Converts the ID to a span ID (if applicable).
241    pub fn as_span_id(&self) -> Option<SpanId> {
242        match self {
243            ParentId::SpanId(span_id) => *span_id,
244        }
245    }
246}

248/// The type of parent entity an attachment is associated with.
249///
250/// This is used to route attachments to different rate limiting buckets, since
251/// depending on the parent the limiting logic is different. E.g. if the attachment has
252/// [`AttachmentParentType::Span`] than it should be dropped if there are span limits.
253///
254/// See [`AttachmentItem::attachment_parent_type`] for how this is determined from an attachment.
255#[derive(Debug)]
256pub enum AttachmentParentType {
257    /// The parent type for all V1 attachments (e.g. minidumps)
258    Event,
259    /// The parent type for all span V2 attachments.
260    Span,
261    /// The parent type for all trace V2 attachments.
262    Trace,
263}
