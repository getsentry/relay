1//! TransactionItem APIs and accounting.
2
3use super::{ItemHeaderKey, TransactionItem};
4use crate::envelope::RejectionQuantities;
5use crate::managed::{Counted, Quantities};
6use relay_quotas::DataCategory;
7use smallvec::smallvec;
8
9impl TransactionItem {
10    /// Returns the declared number of embedded spans, or `None` for an absent or invalid header.
11    pub fn span_count(&self) -> Option<u32> {
12        self.data().headers().get(ItemHeaderKey::SpanCount)
13    }
14
15    /// Counts embedded spans using the header, falling back to scanning the `spans` payload array.
16    /// Invalid payloads count as zero.
17    pub fn count_embedded_spans(&self) -> usize {
18        #[derive(serde::Deserialize)]
19        struct PartialEvent {
20            spans: crate::utils::SeqCount,
21        }
22
23        self.span_count()
24            .map(|count| count as usize)
25            .unwrap_or_else(|| {
26                relay_statsd::metric!(timer(crate::statsd::RelayTimers::CheckNestedSpans), {
27                    serde_json::from_slice::<PartialEvent>(&self.data().payload())
28                        .map_or(0, |event| event.spans.0)
29                })
30            })
31    }
32
33    /// Sets the number of spans in the transaction payload.
34    pub fn set_span_count(&mut self, value: Option<u32>) {
35        self.data_mut()
36            .headers_mut()
37            .set_or_remove(ItemHeaderKey::SpanCount, value);
38    }
39}
40
41impl Counted for TransactionItem {
42    fn quantities(&self) -> Quantities {
43        let spans = 1 + self.span_count().unwrap_or(0) as usize;
44        smallvec![
45            (DataCategory::Transaction, 1),
46            (DataCategory::TransactionIndexed, 1),
47            (DataCategory::Span, spans),
48            (DataCategory::SpanIndexed, spans),
49        ]
50    }
51}
52
53impl RejectionQuantities for TransactionItem {
54    // Include the transaction's own span; the logical transaction is recorded separately.
55    // Read embedded spans from the payload when the count header is absent or invalid.
56    fn rejected_quantities(&self) -> Quantities {
57        let spans = 1 + self.count_embedded_spans();
58        smallvec![
59            (DataCategory::Span, spans),
60            (DataCategory::SpanIndexed, spans),
61        ]
62    }
63}
