1//! Envelope item wire-tag names and conversions.
2
3use std::fmt;
4
5use relay_event_schema::protocol::EventType;
6
7use super::ItemType;

9impl ItemType {
10    /// Returns the event item type corresponding to the given `EventType`.
11    pub fn from_event_type(event_type: EventType) -> Self {
12        match event_type {
13            EventType::Default | EventType::Error => ItemType::Event,
14            EventType::Transaction => ItemType::Transaction,
15            EventType::UserReportV2 => ItemType::UserReportV2,
16            EventType::Csp => ItemType::Security,
17        }
18    }
19
20    /// Returns the variant name of the item type.
21    ///
22    /// Unlike [`Self::as_str`] this returns an unknown value as `unknown`.
23    pub fn name(&self) -> &'static str {
24        match self {
25            Self::Event => "event",
26            Self::Transaction => "transaction",
27            Self::Security => "security",
28            Self::Attachment => "attachment",
29            Self::FormData => "form_data",
30            Self::RawSecurity => "raw_security",
31            Self::UnrealReport => "unreal_report",
32            Self::UserReport => "user_report",
33            Self::UserReportV2 => "feedback",
34            Self::Session => "session",
35            Self::Sessions => "sessions",
36            Self::MetricBuckets => "metric_buckets",
37            Self::ClientReport => "client_report",
38            Self::Profile => "profile",
39            Self::ReplayEvent => "replay_event",
40            Self::ReplayRecording => "replay_recording",
41            Self::ReplayVideo => "replay_video",
42            Self::CheckIn => "check_in",
43            Self::Log => "log",
44            Self::TraceMetric => "trace_metric",
45            Self::Span => "span",
46            Self::ProfileChunk => "profile_chunk",
47            Self::Integration => "integration",
48            Self::Unknown(_) => "unknown",
49        }
50    }
51
52    /// Returns the item type as a string.
53    pub fn as_str(&self) -> &str {
54        match self {
55            Self::Unknown(s) => s,
56            _ => self.name(),
57        }
58    }
59
60    /// Returns `true` if the item is a metric type.
61    pub fn is_metrics(&self) -> bool {
62        matches!(self, ItemType::MetricBuckets)
63    }
64
65    /// Returns `true` if the item is a Relay internal item type.
66    ///
67    /// Internal items are not allowed to be forwarded or sent upstream and will be rejected by
68    /// other Relays. Internal items must be converted to Sentry native items before forwarded.
69    pub fn is_internal(&self) -> bool {
70        matches!(self, ItemType::Integration)
71    }
72}
73
74impl fmt::Display for ItemType {
75    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
76        write!(f, "{}", self.as_str())
77    }
78}

80impl std::str::FromStr for ItemType {
81    type Err = std::convert::Infallible;
82
83    fn from_str(s: &str) -> Result<Self, Self::Err> {
84        Ok(match s {
85            "event" => Self::Event,
86            "transaction" => Self::Transaction,
87            "security" => Self::Security,
88            "attachment" => Self::Attachment,
89            "form_data" => Self::FormData,
90            "raw_security" => Self::RawSecurity,
91            "unreal_report" => Self::UnrealReport,
92            "user_report" => Self::UserReport,
93            "feedback" => Self::UserReportV2,
94            "session" => Self::Session,
95            "sessions" => Self::Sessions,
96            "metric_buckets" => Self::MetricBuckets,
97            "client_report" => Self::ClientReport,
98            "profile" => Self::Profile,
99            "replay_event" => Self::ReplayEvent,
100            "replay_recording" => Self::ReplayRecording,
101            "replay_video" => Self::ReplayVideo,
102            "check_in" => Self::CheckIn,
103            "log" => Self::Log,
104            "trace_metric" => Self::TraceMetric,
105            "span" => Self::Span,
106            "profile_chunk" => Self::ProfileChunk,
107            // "profile_chunk_ui" is to be treated as an alias for `ProfileChunk`
108            // because Android 8.10.0 and 8.11.0 is sending it as the item type.
109            "profile_chunk_ui" => Self::ProfileChunk,
110            "integration" => Self::Integration,
111            other => Self::Unknown(other.to_owned()),
112        })
113    }
114}
