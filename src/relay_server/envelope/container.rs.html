13/// Error emitted when failing to parse an [`ItemContainer`].
14#[derive(thiserror::Error, Debug)]
15pub enum ContainerParseError {
16    /// The item container was expected to have a different content type.
17    #[error("expected item with content type {expected} but got {actual:?}")]
18    MismatchedContentType {
19        expected: ContentType,
20        actual: Option<ContentType>,
21    },
22    /// The item container was expected to have a different item type.
23    #[error("expected item with item type {expected} but got {actual:?}")]
24    MismatchedItemType {
25        expected: ItemType,
26        actual: ItemType,
27    },
28    /// The item container specified length does not match the amount of items contained in the
29    /// container.
30    #[error("container was specified with length {expected:?} but contained {actual} items")]
31    MismatchedLength {
32        expected: Option<u32>,
33        actual: usize,
34    },
35    /// Deserializing items in the container consumed too many operations.
36    #[error("container deserialization exceeded operations limit {limit}")]
37    LimitExceeded { limit: usize },
38    /// The container is malformed and cannot be deserialized.
39    #[error("failed to deserialize item container: {0}")]
40    Deserialize(#[from] serde_json::Error),
41}

43/// Error emitted when failing to write/serialize and [`ItemContainer`].
44#[derive(thiserror::Error, Debug)]
45pub enum ContainerWriteError {
46    /// The item container is too large to serialize.
47    #[error("failed to serialize item container, item count overflow")]
48    Overflow,
49    /// The contained items cannot be serialized.
50    #[error("failed to serialize item container: {0}")]
51    Serialize(#[from] serde_json::Error),
52}

59/// Any item contained in an [`ItemContainer`] needs to implement this trait.
60pub trait ContainerItem: FromValue + IntoValue {
61    /// The expected item type of the container for this type.
62    const ITEM_TYPE: ItemType;
63    /// The expected content type of the container for this type.
64    const CONTENT_TYPE: ContentType;
65
66    /// [`ItemContainer`] metadata.
67    ///
68    /// Metadata which is associated with the item container itself.
69    ///
70    /// This may be used for supplying additional processing hints and versioning.
71    type Metadata: DeserializeOwned + Serialize + std::fmt::Debug;
72
73    /// Header associated with the item.
74    ///
75    /// The header will be automatically serialized and de-serialized by the [`ItemContainer`].
76    /// All headers must be de-serialized from and into an object keyed with a string.
77    ///
78    /// To ensure compatibility, Relay should almost always make sure each header is optional
79    /// or has a default defined.
80    ///
81    /// Use [`NoHeader`] when there are no explicit headers defined.
82    type Header: DeserializeOwned + Serialize + std::fmt::Debug;
83}

85/// A metadata implementation for [`container items`](ContainerItem::Metadata) which currently do
86/// not have any metadata defined.
87///
88/// This implementation discards all metadata and does not provide forward compatibility, as this
89/// may not always be desired.
90/// For example an ingestion setting like `infer client ip`, must be handled by the first Relay
91/// instance and should  not be forwarded to a later Relay instance.
92///
93/// If forward compatibility is a concern, a [`ContainerItem`] should provide its own forward
94/// compatible type with the appropriate handling.
95#[derive(Debug, Default, Copy, Clone)]
96pub struct NoMetadata;

98impl<'de> Deserialize<'de> for NoMetadata {
99    fn deserialize<D>(deserializer: D) -> Result<Self, D::Error>
100    where
101        D: serde::Deserializer<'de>,
102    {
103        serde::de::IgnoredAny::deserialize(deserializer)?;
104        Ok(NoMetadata)
105    }
106}

110    where
111        S: serde::Serializer,
112    {
113        serializer.serialize_map(Some(0))?.end()
114    }

126pub struct WithHeader<T: ContainerItem> {

128    pub header: Option<T::Header>,

133impl<T: ContainerItem> WithHeader<T> {
134    /// Creates a [`Self`] from just a value with no associated header.
135    ///
136    /// This should only be used when creating new items, in most cases existing headers should be
137    /// respected and explicitly handled and passed along.
138    ///
139    /// Prefer using [`WithHeader::new`] where possible.
140    pub fn just(value: Annotated<T>) -> Self {
141        Self {
142            header: None,
143            value,
144        }
145    }

152    /// set by only implementing [`Self::new`] for items which have a [`NoHeader`] as header.

172impl<'de, T: ContainerItem> Deserialize<'de> for WithHeader<T> {
173    fn deserialize<D>(deserializer: D) -> Result<Self, D::Error>
174    where
175        D: serde::Deserializer<'de>,
176    {
177        #[derive(Deserialize)]
178        #[serde(bound(deserialize = "T: ContainerItem"))]
179        struct Inner<T: ContainerItem> {
180            #[serde(rename = "__header")]
181            header: Option<T::Header>,
182            #[serde(flatten)]
183            value: DeserializableAnnotated<T>,
184        }
185
186        let Inner { header, value } = Inner::<T>::deserialize(deserializer)?;
187        Ok(Self {
188            header,
189            value: value.0,
190        })
191    }
192}

194impl<T: ContainerItem> Serialize for WithHeader<T> {
195    fn serialize<S>(&self, serializer: S) -> Result<S::Ok, S::Error>
196    where
197        S: ser::Serializer,
198    {
199        #[derive(Serialize)]
200        #[serde(bound(serialize = "T: ContainerItem"))]
201        struct Inner<'a, T: ContainerItem> {
202            #[serde(rename = "__header", skip_serializing_if = "Option::is_none")]
203            header: Option<&'a T::Header>,
204            #[serde(flatten)]
205            value: SerializableAnnotated<'a, T>,
206        }
207

210            value: SerializableAnnotated(&self.value),
211        };

214    }

220/// A container for multiple homogeneous envelope items.
221///
222/// Item containers are used to minimize the amount of single envelope items contained in an
223/// envelope. They massively improve parsing speed of envelopes in Relay but are also used
224/// to minimize metadata duplication on item headers.
225///
226/// Especially for small envelope items with high quantities (e.g. logs), this drastically
227/// improves fast path parsing speeds and minimizes serialization overheads, by minimizing
228/// the amount of items in an envelope.
229///
230/// An item container does not have a special [`super::ItemType`], but is identified by the
231/// content type of the item.
232#[derive(Debug)]
233pub struct ItemContainer<T: ContainerItem> {
234    metadata: Result<T::Metadata, InvalidMetadata>,
235    items: ContainerItems<T>,
236}
237
238impl<T: ContainerItem> ItemContainer<T> {
239    /// Creates a new [`ItemContainer`] with the given metadata and items.
240    pub fn from_parts(metadata: T::Metadata, items: ContainerItems<T>) -> Self {
241        Self {
242            metadata: Ok(metadata),
243            items,
244        }
245    }
246
247    /// Consumes the container returning the metadata and items parts.
248    pub fn into_parts(self) -> (Result<T::Metadata, InvalidMetadata>, ContainerItems<T>) {
249        (self.metadata, self.items)
250    }
251
252    /// Parses an [`ItemContainer`] from an envelope [`Item`].
253    ///
254    /// This function also validates metadata of the container, specifically the content type
255    /// and amount of contained items.
256    pub fn parse(item: &Item, max_ops: usize) -> Result<Self, ContainerParseError> {
257        if item.content_type() != Some(T::CONTENT_TYPE) {
258            return Err(ContainerParseError::MismatchedContentType {
259                expected: T::CONTENT_TYPE,
260                actual: item.content_type(),
261            });
262        }
263
264        if item.ty() != &T::ITEM_TYPE {
265            return Err(ContainerParseError::MismatchedItemType {
266                expected: T::ITEM_TYPE,
267                actual: item.ty().clone(),
268            });
269        }
270
271        let payload = item.payload();
272        // Currently we assume every payload is JSON, but in the future we may allow other formats.
273        let mut de = serde_json::Deserializer::from_slice(&payload);
274        let container = Self::deserialize(&mut de, max_ops).map_err(|e| match e {
275            relay_serialization::serde::Error::LimitExceeded(limit) => {
276                ContainerParseError::LimitExceeded { limit }
277            }
278            relay_serialization::serde::Error::Serde(e) => ContainerParseError::Deserialize(e),
279        })?;
280
281        if Some(container.items.len()) != item.item_count().map(|u| u as usize) {
282            return Err(ContainerParseError::MismatchedLength {
283                expected: item.item_count(),
284                actual: container.items.len(),
285            });
286        }
287
288        Ok(container)
289    }
290
291    /// Serializes the [`ItemContainer`] into an envelope [`Item`].
292    ///
293    /// This will serialize the contained items into the [`Item::payload`] as well as
294    /// update the [Item::content_type] and [`Item::item_count`].
295    pub fn write_to(&self, item: &mut Item) -> Result<(), ContainerWriteError> {
296        let mut payload = bytes::BytesMut::with_capacity(256).writer();
297        let mut ser = serde_json::Serializer::new(&mut payload);
298
299        self.serialize(&mut ser)?;
300
301        item.set_payload_with_item_count(
302            T::CONTENT_TYPE,
303            payload.into_inner(),
304            u32::try_from(self.items.len()).map_err(|_| ContainerWriteError::Overflow)?,
305        );
306
307        Ok(())
308    }
309
310    /// Checks whether a given item is a container of `T`s, according to its item and content types.
311    pub fn is_container(item: &Item) -> bool {
312        item.ty() == &T::ITEM_TYPE && item.content_type() == Some(T::CONTENT_TYPE)
313    }
314
315    fn deserialize<'de, D: de::Deserializer<'de>>(
316        deserializer: D,
317        max_ops: usize,
318    ) -> Result<Self, relay_serialization::serde::Error<D::Error>> {
319        struct TryMetadata<T: ContainerItem>(Result<T::Metadata, InvalidMetadata>);
320
321        impl<'de, T> Deserialize<'de> for TryMetadata<T>
322        where
323            T: ContainerItem,
324        {
325            fn deserialize<D>(deserializer: D) -> Result<Self, D::Error>
326            where
327                D: serde::Deserializer<'de>,
328            {
329                #[derive(Deserialize)]
330                #[serde(untagged)]
331                enum Try<T> {
332                    T(T),
333                    Err(serde::de::IgnoredAny),
334                }
335
336                Ok(match Try::<T::Metadata>::deserialize(deserializer)? {
337                    Try::T(t) => TryMetadata(Ok(t)),
338                    Try::Err(_) => TryMetadata(Err(InvalidMetadata(()))),
339                })
340            }
341        }
342
343        #[derive(Deserialize)]
344        #[serde(bound(deserialize = "T: ContainerItem"))]
345        struct Layout<T: ContainerItem> {
346            items: ContainerItems<T>,
347            #[serde(flatten)]
348            metadata: TryMetadata<T>,
349        }
350
351        let Layout {
352            items,
353            metadata: TryMetadata(metadata),
354        } = relay_serialization::serde::deserialize(deserializer, max_ops)?;
355
356        Ok(Self { items, metadata })
357    }
358
359    fn serialize<S: ser::Serializer>(&self, serializer: S) -> Result<S::Ok, S::Error> {
360        #[derive(Serialize)]
361        #[serde(bound(serialize = "T: ContainerItem"))]
362        struct Layout<'a, T: ContainerItem> {
363            #[serde(flatten)]
364            metadata: Option<&'a T::Metadata>,
365            items: &'a ContainerItems<T>,
366        }
367
368        let layout = Layout {
369            metadata: self.metadata.as_ref().ok(),
370            items: &self.items,
371        };
372        Serialize::serialize(&layout, serializer)
373    }
374}

378        Self {

380            metadata: Ok(NoMetadata),
381        }
382    }
