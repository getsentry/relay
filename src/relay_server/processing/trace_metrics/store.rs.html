25macro_rules! required {
26    ($value:expr) => {{
27        match $value {
28            Annotated(Some(value), _) => value,
29            Annotated(None, meta) => {
30                relay_log::debug!(
31                    "dropping trace metric because of missing required field {} with meta {meta:?}",
32                    stringify!($value),
33                );
34                return Err(Error::Invalid(DiscardReason::InvalidTraceMetric));
35            }
36        }
37    }};
38}

60        let Some(attrs) = span.attributes.value() else {
61            return Self::default();
62        };

64        Self {
65            client: extract_client_sample_rate(attrs),
66            server: extract_server_sample_rate(attrs),
67        }
68    }

71pub fn convert(metric: WithHeader<TraceMetric>, ctx: &Context) -> Result<StoreTraceItem> {
72    let quantities = metric.quantities();
73    let payload_size_bytes = metric
74        .header
75        .as_ref()
76        .and_then(|h| h.byte_size)
77        .unwrap_or_default();
78
79    let metric = required!(metric.value);
80    let timestamp = required!(metric.timestamp);
81
82    let meta = extract_meta_attributes(&metric, &metric.attributes);
83    let attrs = metric.attributes.0.unwrap_or_default();
84    let fields = FieldAttributes {
85        metric_name: required!(metric.name),
86        metric_type: required!(metric.ty),
87        metric_unit: metric.unit.into_value(),
88        value: extract_numeric_value(required!(metric.value))?,
89        timestamp,
90        span_id: metric.span_id.into_value(),
91        payload_size_bytes,
92    };
93
94    // Derived metrics inherit the sample rates of their source span.
95    let derived_sample_rates = ctx.derived_sample_rates.unwrap_or_default();
96    let client_sample_rate = derived_sample_rates
97        .client
98        .or_else(|| extract_client_sample_rate(&attrs))
99        .unwrap_or(1.0);
100    let server_sample_rate = derived_sample_rates.server.unwrap_or(1.0);
101
102    let trace_item = TraceItem {
103        item_type: TraceItemType::Metric.into(),
104        organization_id: ctx.scoping.organization_id.value(),
105        project_id: ctx.scoping.project_id.value(),
106        received: Some(ts(ctx.received_at)),
107        retention_days: ctx.retention.standard.into(),
108        downsampled_retention_days: ctx.retention.downsampled.into(),
109        timestamp: Some(ts(timestamp.0)),
110        trace_id: required!(metric.trace_id).to_string(),
111        item_id: uuid_to_item_id(Uuid::new_v7(timestamp.into())),
112        attributes: attributes(meta, attrs, fields),
113        client_sample_rate,
114        server_sample_rate,
115        outcomes: Some(quantities_to_trace_item_outcomes(quantities, ctx.scoping)),
116    };
117
118    Ok(StoreTraceItem { trace_item })
119}

121fn ts(dt: DateTime<Utc>) -> Timestamp {
122    Timestamp {
123        seconds: dt.timestamp(),
124        nanos: i32::try_from(dt.timestamp_subsec_nanos()).unwrap_or(0),
125    }

128struct FieldAttributes {
129    metric_name: String,
130    metric_type: MetricType,
131    metric_unit: Option<MetricUnit>,
132    value: f64,
133    timestamp: relay_event_schema::protocol::Timestamp,
134    span_id: Option<SpanId>,
135    payload_size_bytes: u64,
136}

138fn extract_numeric_value(value: Value) -> Result<f64> {
139    match value {
140        Value::F64(v) => Ok(v),
141        Value::I64(v) => Ok(v as f64),
142        Value::U64(v) => Ok(v as f64),
143        _ => Err(Error::Invalid(DiscardReason::InvalidTraceMetric)),
144    }
145}

147fn attributes(
148    meta: HashMap<String, AnyValue>,
149    attributes: Attributes,
150    fields: FieldAttributes,
151) -> HashMap<String, AnyValue> {
152    let mut result = meta;
153    // +N, one for each field attribute added and some extra for potential meta.
154    result.reserve(attributes.0.len() + 15);
155
156    processing::utils::store::convert_attributes_into(&mut result, attributes);
157
158    let FieldAttributes {
159        metric_name,
160        metric_type,
161        metric_unit,
162        value,
163        timestamp,
164        span_id,
165        payload_size_bytes,
166    } = fields;
167
168    result.insert(
169        "sentry.metric_name".to_owned(),
170        AnyValue {
171            value: Some(any_value::Value::StringValue(metric_name.clone())),
172        },
173    );
174
175    result.insert(
176        "sentry.metric_type".to_owned(),
177        AnyValue {
178            value: Some(any_value::Value::StringValue(metric_type.to_string())),
179        },
180    );
181
182    // Add key names matching metric name and type to workaround current co-occuring attributes limitations.
183    result.insert(
184        format!("sentry._internal.cooccuring.name.{metric_name}"),
185        AnyValue {
186            value: Some(any_value::Value::BoolValue(true)),
187        },
188    );
189    result.insert(
190        format!("sentry._internal.cooccuring.type.{metric_type}"),
191        AnyValue {
192            value: Some(any_value::Value::BoolValue(true)),
193        },
194    );
195
196    if let Some(metric_unit) = metric_unit {
197        result.insert(
198            "sentry.metric_unit".to_owned(),
199            AnyValue {
200                value: Some(any_value::Value::StringValue(metric_unit.to_string())),
201            },
202        );
203
204        result.insert(
205            format!("sentry._internal.cooccuring.unit.{metric_unit}"),
206            AnyValue {
207                value: Some(any_value::Value::BoolValue(true)),
208            },
209        );
210    }
211
212    result.insert(
213        "sentry.value".to_owned(),
214        AnyValue {
215            value: Some(any_value::Value::DoubleValue(value)),
216        },
217    );
218
219    let timestamp_nanos = timestamp
220        .into_inner()
221        .timestamp_nanos_opt()
222        .unwrap_or_default();
223
224    result.insert(
225        "sentry.timestamp_precise".to_owned(),
226        AnyValue {
227            value: Some(any_value::Value::IntValue(timestamp_nanos)),
228        },
229    );
230
231    if let Some(span_id) = span_id {
232        result.insert(
233            "sentry.span_id".to_owned(),
234            AnyValue {
235                value: Some(any_value::Value::StringValue(span_id.to_string())),
236            },
237        );
238    }
239
240    result.insert(
241        "sentry.payload_size_bytes".to_owned(),
242        AnyValue {
243            value: Some(any_value::Value::IntValue(payload_size_bytes as i64)),
244        },
245    );
246
247    result
248}

250/// Produce the supplied webvital trace metrics to kafka.
251/// This is required right now to double-write these webvitals as trace metrics, while we
252/// still write the vitals as spans.  Eventually, the sdks will natively emit metrics and we
253/// can remove this code.
254pub fn produce_webvitals_metrics(
255    s: processing::StoreHandle<'_>,
256    span: &Managed<Box<crate::services::store::StoreSpanV2>>,
257    metrics: Vec<TraceMetric>,
258) {
259    for metric in metrics {
260        let trace_metric_headers = trace_metric::TraceMetricHeader {
261            byte_size: Some(utils::calculate_size(&metric)),
262            other: std::collections::BTreeMap::default(),
263        };
264
265        let wheader = crate::envelope::WithHeader {
266            header: trace_metric_headers.into(),
267            value: metric.into(),
268        };
269
270        if let Ok(mut item) = store::convert(
271            wheader,
272            &store::Context {
273                received_at: span.received_at(),
274                scoping: span.scoping(),
275                retention: processing::Retention {
276                    standard: span.retention_days,
277                    downsampled: span.downsampled_retention_days,
278                },
279                // Carry sample rates from the origin span so that metrics can be extrapolated.
280                derived_sample_rates: Some(SampleRates::from_span(&span.item)),
281            },
282        ) {
283            // Clear outcomes for these metrics, as we don't want them billed.
284            item.trace_item.outcomes = None;
285            s.send_to_store(span.wrap(item));
286        }
287    }
288}

439    fn test_sample_rates_from_span() {
440        let mut attrs = BTreeMap::new();
441        attrs.insert(
442            "sentry.client_sample_rate".to_owned(),
443            sample_rate_attribute(0.5),
444        );
445        attrs.insert(
446            "sentry.server_sample_rate".to_owned(),
447            sample_rate_attribute(0.01),
448        );
449        let span = SpanV2 {
450            attributes: Annotated::new(Attributes(attrs)),
451            ..Default::default()
452        };
453
454        let rates = SampleRates::from_span(&span);
455        assert_eq!(rates.client, Some(0.5));
456        assert_eq!(rates.server, Some(0.01));
457
458        // Spans without sample rates leave the metric unweighted.
459        let rates = SampleRates::from_span(&SpanV2::default());
460        assert_eq!(rates.client, None);
461        assert_eq!(rates.server, None);
462    }

464    #[test]
465    fn test_extract_client_sample_rate_function() {
466        let mut attrs_map = BTreeMap::new();
467        let attr = Attribute {
468            value: AttributeValue {
469                ty: Annotated::new(AttributeType::Double),
470                value: Annotated::new(Value::F64(0.5)),
471            },
472            other: Object::new(),
473        };
474        attrs_map.insert("sentry.client_sample_rate".to_owned(), Annotated::new(attr));
475        let attrs = Attributes(attrs_map);
476
477        let sample_rate = extract_client_sample_rate(&attrs);
478        assert_eq!(sample_rate, Some(0.5));
479
480        // Without sample rate attribute
481        let empty_attrs = Attributes(BTreeMap::new());
482        let default_rate = extract_client_sample_rate(&empty_attrs);
483        assert_eq!(default_rate, None);
484
485        // Invalid sample rate
486        let invalid_attrs = Attributes(BTreeMap::from([(
487            "sentry.client_sample_rate".to_owned(),
488            Annotated::new(Attribute {
489                value: AttributeValue {
490                    ty: Annotated::new(AttributeType::Double),
491                    value: Annotated::new(Value::F64(2.0)),
492                },
493                other: Object::new(),
494            }),
495        )]));
496        let invalid_rate = extract_client_sample_rate(&invalid_attrs);
497        assert_eq!(invalid_rate, None);
498    }
