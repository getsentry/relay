13/// Creates a span from the transaction and applies tag extraction on it.
14///
15/// Returns `None` when [`tag_extraction::extract_span_tags`] clears the span, which it shouldn't.
16pub fn extract_segment_span(
17    event: &Event,
18    max_tag_value_size: usize,
19    span_allowed_hosts: &[String],
20) -> Option<Span> {
21    let mut spans = [Span::from(event).into()];
22
23    tag_extraction::extract_span_tags(event, &mut spans, max_tag_value_size, span_allowed_hosts);
24    tag_extraction::extract_segment_span_tags(event, &mut spans);
25
26    spans.into_iter().next().and_then(Annotated::into_value)
27}

29/// Input arguments for [`extract_metrics`].
30pub struct ExtractMetricsContext<'a> {
31    pub config: CombinedMetricExtractionConfig<'a>,
32    pub dsc: Option<&'a DynamicSamplingContext>,
33    pub project_id: ProjectId,
34    pub ctx: Context<'a>,
35    pub sampling_decision: SamplingDecision,
36    pub extract_span_metrics: bool,

39/// Extracts metrics from a transaction and its spans.
40pub fn extract_metrics(
41    event: &mut Annotated<Event>,
42    extracted_metrics: &mut ProcessingExtractedMetrics,
43    ctx: ExtractMetricsContext,
44) {
45    let ExtractMetricsContext {
46        config,
47        dsc,
48        project_id,
49        ctx,
50        sampling_decision,
51        extract_span_metrics,
52    } = ctx;
53    let Some(event) = event.value_mut() else {
54        // Nothing to extract.
55        return;
56    };
57
58    let metrics = crate::metrics_extraction::event::extract_metrics(
59        event,
60        crate::metrics_extraction::event::ExtractMetricsConfig {
61            config,
62            sampling_decision,
63            target_project_id: project_id,
64            max_tag_value_size: ctx
65                .config
66                .aggregator_config_for(MetricNamespace::Spans)
67                .max_tag_value_length,
68            extract_spans: extract_span_metrics,
69            transaction_from_dsc: dsc.and_then(|dsc| dsc.transaction.as_deref()),
70        },
71    );
72    extracted_metrics.extend(metrics, Some(sampling_decision));
73}
