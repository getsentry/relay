25/// A processor for OTEL log integrations.
26///
27/// This processor only produces intermediates (items
28/// that are put back into the processing pipeline) by
29/// converting log integration items into actual logs. It
30/// never passes anything on to the store or another Relay.
31#[derive(Debug)]
32pub struct OtelLogsIntegrationProcessor;
33
34impl processing::Processor for OtelLogsIntegrationProcessor {
35    type Input = SerializedOtelLogsIntegration;
36    type Output = Nothing;
37    type Error = Error;
38
39    fn cogs() -> relay_cogs::FeatureWeights {
40        AppFeature::Logs.into()
41    }
42
43    fn prepare_envelope(&self, envelope: &mut ManagedEnvelope) -> Option<Managed<Self::Input>> {
44        let (item, format) =
45            envelope
46                .envelope_mut()
47                .take_item_by_map(|item| match item.integration() {
48                    Some(Integration::Logs(LogsIntegration::OtelV1 { format })) => Some(format),
49                    _ => None,
50                })?;
51        let headers = envelope.envelope().headers().clone();
52
53        let work =
54            SerializedOtelLogsIntegration(SerializedLogsIntegration { headers, item }, format);
55        Some(Managed::with_meta_from_managed_envelope(envelope, work))
56    }
57
58    async fn process(
59        &self,
60        work: Managed<Self::Input>,
61        ctx: processing::Context<'_>,
62    ) -> Result<Output<Self::Output>, Rejected<Self::Error>> {
63        let intermediates = work.try_map(|work, records| {
64            expand(work, records, ctx.config.max_expanded_log_operations())
65        })?;
66        Ok(Output::intermediates(intermediates))
67    }
68}

70/// Expands a OTEL log integration into an envelope of logs that can be
71/// returned to the processing pipeline.
72pub fn expand(
73    SerializedOtelLogsIntegration(SerializedLogsIntegration { headers, item }, format): SerializedOtelLogsIntegration,
74    records: &mut RecordKeeper,
75    max_ops: usize,
76) -> Result<Intermediates, Error> {
77    // Log byte counts will change here as we go from an estimated count based on the body, to
78    // accurately counted bytes.
79    records.lenient(DataCategory::LogByte);
80
81    let payload = item.payload();
82    let mut logs = Vec::new();
83    let parsed = parse_logs_data(format, &payload, max_ops)?;
84
85    for resource_logs in parsed.resource_logs {
86        let resource = resource_logs.resource.as_ref();
87        for scope_logs in resource_logs.scope_logs {
88            let scope = scope_logs.scope.as_ref();
89            for log_record in scope_logs.log_records {
90                let log = relay_ourlogs::otel_to_sentry_log(log_record, resource, scope);
91                // https://github.com/getsentry/relay/issues/6605: This is only used to keep records in case serializing
92                // the logs back into an envelope fails. Consider removing it.
93                let byte_size = relay_ourlogs::calculate_size(&log);
94
95                records.add([
96                    (DataCategory::LogItem, 1),
97                    (DataCategory::LogByte, byte_size as usize),
98                ]);
99
100                logs.push(WithHeader::just(log.into()));
101            }
102        }
103    }
104
105    // Undo all the base item quantities, as they will be completely taken over by the parsed
106    // contents, which contains an arbitrary amount of items (even 0).
107    records.remove(&item);
108
109    let settings = IngestSettings {
110        infer_user_agent: Some(InferUserAgent::Never),
111        infer_ip: Some(InferIp::Never),
112    };
113
114    create_intermediates(headers, logs, settings)
115}

117fn parse_logs_data(format: OtelFormat, payload: &[u8], max_ops: usize) -> Result<LogsData, Error> {
118    match format {
119        OtelFormat::Json => {
120            let mut de = serde_json::Deserializer::from_reader(payload);
121            let result = relay_serialization::serde::deserialize(&mut de, max_ops).map_err(|e| {
122                relay_log::debug!(
123                    error = &e as &dyn std::error::Error,
124                    "Failed to parse logs data as JSON"
125                );
126                Error::Invalid(DiscardReason::InvalidJson)
127            });
128            de.end()
129                .map_err(|_| Error::Invalid(DiscardReason::InvalidJson))?;
130            result
131        }
132        OtelFormat::Protobuf => LogsData::decode_bounded(payload, max_ops).map_err(|e| {
133            relay_log::debug!(
134                error = &e as &dyn std::error::Error,
135                "Failed to parse logs data as protobuf"
136            );
137            Error::Invalid(DiscardReason::InvalidProtobuf)
138        }),
139    }

314    #[test]
315    fn test_protobuf_and_json_limits_agree() {
316        // The same logical payload is accepted or rejected regardless of its encoding.
317        let small = logs_with_records(10);
318        let large = logs_with_records(2_000);
319
320        for (logs, ok) in [(small, true), (large, false)] {
321            let proto = parse_logs_data(OtelFormat::Protobuf, &logs.encode_to_vec(), 1000);
322            let json = parse_logs_data(
323                OtelFormat::Json,
324                serde_json::to_string(&logs).unwrap().as_bytes(),
325                1000,
326            );
327            assert_eq!(proto.is_ok(), ok);
328            assert_eq!(json.is_ok(), ok);
329        }
330    }
