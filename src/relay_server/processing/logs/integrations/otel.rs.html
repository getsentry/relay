25/// A processor for OTEL log integrations.
26///
27/// This processor only produces intermediates (items
28/// that are put back into the processing pipeline) by
29/// converting log integration items into actual logs. It
30/// never passes anything on to the store or another Relay.
31#[derive(Debug)]
32pub struct OtelLogsIntegrationProcessor;
33
34impl processing::Processor for OtelLogsIntegrationProcessor {
35    type Input = SerializedOtelLogsIntegration;
36    type Output = Nothing;
37    type Error = Error;
38
39    fn cogs() -> relay_cogs::FeatureWeights {
40        AppFeature::Logs.into()
41    }
42
43    fn prepare_envelope(&self, envelope: &mut ManagedEnvelope) -> Option<Managed<Self::Input>> {
44        let meta = envelope.envelope().headers().meta().clone();
45
46        let (item, format) =
47            envelope
48                .envelope_mut()
49                .take_item_by_map(|item| match item.integration() {
50                    Some(Integration::Logs(LogsIntegration::OtelV1 { format })) => Some(format),
51                    _ => None,
52                })?;
53
54        let work = SerializedOtelLogsIntegration(SerializedLogsIntegration { meta, item }, format);
55        Some(Managed::with_meta_from_managed_envelope(envelope, work))
56    }
57
58    async fn process(
59        &self,
60        work: Managed<Self::Input>,
61        _ctx: processing::Context<'_>,
62    ) -> Result<Output<Self::Output>, Rejected<Self::Error>> {
63        let intermediates = work.try_map(expand)?;
64        Ok(Output::intermediates(intermediates))
65    }
66}

68/// Expands a OTEL log integration into an envelope of logs that can be
69/// returned to the processing pipeline.
70pub fn expand(
71    work: SerializedOtelLogsIntegration,
72    records: &mut RecordKeeper,
73) -> Result<Intermediates, Error> {
74    let SerializedOtelLogsIntegration(SerializedLogsIntegration { meta, item }, format) = work;
75
76    // Log byte counts will change here as we go from an estimated count based on the body, to
77    // accurately counted bytes.
78    records.lenient(DataCategory::LogByte);
79
80    let payload = item.payload();
81    let mut logs = Vec::new();
82    let parsed = parse_logs_data(format, &payload)?;
83
84    for resource_logs in parsed.resource_logs {
85        let resource = resource_logs.resource.as_ref();
86        for scope_logs in resource_logs.scope_logs {
87            let scope = scope_logs.scope.as_ref();
88            for log_record in scope_logs.log_records {
89                let log = relay_ourlogs::otel_to_sentry_log(log_record, resource, scope);
90                // INGEST-1333: This is only used to keep records in case serializing
91                // the logs back into an envelope fails. Consider removing it.
92                let byte_size = relay_ourlogs::calculate_size(&log);
93
94                records.modify_by(DataCategory::LogItem, 1);
95                records.modify_by(DataCategory::LogByte, byte_size as isize);
96
97                logs.push(WithHeader::just(log.into()));
98            }
99        }
100    }
101
102    // Undo all the base item quantities, as they will be completely taken over by the parsed
103    // contents, which contains an arbitrary amount of items (even 0).
104    for (category, quantity) in item.quantities() {
105        records.modify_by(category, -(quantity as isize));
106    }
107
108    let settings = IngestSettings {
109        infer_user_agent: Some(InferUserAgent::Never),
110        infer_ip: Some(InferIp::Never),
111    };
112
113    create_intermediates(meta, logs, settings)
114}

116fn parse_logs_data(format: OtelFormat, payload: &[u8]) -> Result<LogsData, Error> {
117    match format {
118        OtelFormat::Json => serde_json::from_slice(payload).map_err(|e| {
119            relay_log::debug!(
120                error = &e as &dyn std::error::Error,
121                "Failed to parse logs data as JSON"
122            );
123            Error::Invalid(DiscardReason::InvalidJson)
124        }),
125        OtelFormat::Protobuf => LogsData::decode(payload).map_err(|e| {
126            relay_log::debug!(
127                error = &e as &dyn std::error::Error,
128                "Failed to parse logs data as protobuf"
129            );
130            Error::Invalid(DiscardReason::InvalidProtobuf)
131        }),
132    }
133}
