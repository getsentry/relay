24/// A processor for Vercel log integrations.
25///
26/// This processor only produces intermediates (items
27/// that are put back into the processing pipeline) by
28/// converting log integration items into actual logs. It
29/// never passes anything on to the store or another Relay.
30#[derive(Debug)]
31pub struct VercelLogsIntegrationProcessor;
32
33impl processing::Processor for VercelLogsIntegrationProcessor {
34    type Input = SerializedVercelLogsIntegration;
35    type Output = Nothing;
36    type Error = Error;
37
38    fn cogs() -> relay_cogs::FeatureWeights {
39        AppFeature::Logs.into()
40    }
41
42    fn prepare_envelope(&self, envelope: &mut ManagedEnvelope) -> Option<Managed<Self::Input>> {
43        let meta = envelope.envelope().headers().meta().clone();
44
45        let (item, format) =
46            envelope
47                .envelope_mut()
48                .take_item_by_map(|item| match item.integration() {
49                    Some(Integration::Logs(LogsIntegration::VercelDrainLog { format })) => {
50                        Some(format)
51                    }
52                    _ => None,
53                })?;
54
55        let work =
56            SerializedVercelLogsIntegration(SerializedLogsIntegration { meta, item }, format);
57        Some(Managed::with_meta_from_managed_envelope(envelope, work))
58    }
59
60    async fn process(
61        &self,
62        work: Managed<Self::Input>,
63        _ctx: processing::Context<'_>,
64    ) -> Result<Output<Self::Output>, Rejected<Self::Error>> {
65        let intermediates = work.try_map(expand)?;
66        Ok(Output::intermediates(intermediates))
67    }
68}

70/// Expands a Vercel log integration into an envelope of logs that can be
71/// returned to the processing pipeline.
72pub fn expand(
73    work: SerializedVercelLogsIntegration,
74    records: &mut RecordKeeper,
75) -> Result<Intermediates, Error> {
76    let SerializedVercelLogsIntegration(SerializedLogsIntegration { meta, item }, format) = work;
77
78    // Log byte counts will change here as we go from an estimated count based on the body, to
79    // accurately counted bytes.
80    records.lenient(DataCategory::LogByte);
81
82    let mut logs = Vec::new();
83    let mut produce = |log| {
84        // INGEST-1333: This is only used to keep records in case serializing
85        // the logs back into an envelope fails. Consider removing it.
86        let byte_size = relay_ourlogs::calculate_size(&log);
87
88        records.modify_by(DataCategory::LogItem, 1);
89        records.modify_by(DataCategory::LogByte, byte_size as isize);
90
91        logs.push(WithHeader::just(log.into()));
92    };
93
94    let mut count: i32 = 0;
95
96    let payload = item.payload();
97    match format {
98        VercelLogDrainFormat::Json => {
99            let logs = serde_json::from_slice::<Vec<VercelLog>>(&payload).map_err(|e| {
100                relay_log::debug!(
101                    error = &e as &dyn std::error::Error,
102                    "Failed to parse logs data as JSON"
103                );
104                Error::Invalid(DiscardReason::InvalidJson)
105            })?;
106
107            for log in logs {
108                count += 1;
109                let ourlog = relay_ourlogs::vercel_log_to_sentry_log(log);
110                produce(ourlog);
111            }
112        }
113        VercelLogDrainFormat::NdJson => {
114            for line in payload.split(|&b| b == b'\n') {
115                if line.is_empty() {
116                    continue;
117                }
118
119                if let Ok(log) = serde_json::from_slice::<VercelLog>(line) {
120                    count += 1;
121                    let ourlog = relay_ourlogs::vercel_log_to_sentry_log(log);
122                    produce(ourlog);
123                }
124            }
125        }
126    }
127
128    // Undo all the base item quantities, as they will be completely taken over by the parsed
129    // contents, which contains an arbitrary amount of items (even 0).
130    for (category, quantity) in item.quantities() {
131        records.modify_by(category, -(quantity as isize));
132    }
133
134    if count == 0 {
135        relay_log::debug!("Failed to parse any logs from vercel log drain payload");
136        return Err(Error::Invalid(DiscardReason::InvalidJson));
137    }
138
139    let settings = IngestSettings {
140        infer_user_agent: Some(InferUserAgent::Never),
141        infer_ip: Some(InferIp::Never),
142    };
143
144    create_intermediates(meta, logs, settings)
145}
