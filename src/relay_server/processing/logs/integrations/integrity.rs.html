16/// A processor for Integrity Report log integrations.
17///
18/// This processor only produces intermediates (items
19/// that are put back into the processing pipeline) by
20/// converting log integration items into actual logs. It
21/// never passes anything on to the store or another Relay.
22#[derive(Debug)]
23pub struct IntegrityLogsIntegrationProcessor;
24
25impl processing::Processor for IntegrityLogsIntegrationProcessor {
26    type Input = SerializedLogsIntegration;
27    type Output = Nothing;
28    type Error = Error;
29
30    fn cogs() -> relay_cogs::FeatureWeights {
31        AppFeature::Logs.into()
32    }
33
34    fn prepare_envelope(&self, envelope: &mut ManagedEnvelope) -> Option<Managed<Self::Input>> {
35        let meta = envelope.envelope().headers().meta().clone();
36
37        let item = envelope.envelope_mut().take_item_by(|item| {
38            matches!(
39                item.integration(),
40                Some(Integration::Logs(LogsIntegration::Integrity))
41            )
42        })?;
43
44        let work = SerializedLogsIntegration { meta, item };
45        Some(Managed::with_meta_from_managed_envelope(envelope, work))
46    }
47
48    async fn process(
49        &self,
50        work: Managed<Self::Input>,
51        _ctx: processing::Context<'_>,
52    ) -> Result<Output<Self::Output>, Rejected<Self::Error>> {
53        let intermediates = work.try_map(expand)?;
54        Ok(Output::intermediates(intermediates))
55    }
56}
57
58/// Expands an Integrity Report log integration into an envelope of logs that can be
59/// returned to the processing pipeline.
60pub fn expand(
61    work: SerializedLogsIntegration,
62    records: &mut RecordKeeper,
63) -> Result<Intermediates, Error> {
64    let SerializedLogsIntegration { meta, item } = work;
65
66    // Log byte counts will change here as we go from an estimated count based on the body, to
67    // accurately counted bytes.
68    records.lenient(DataCategory::LogByte);
69
70    let payload = item.payload();
71    let received_at = meta.received_at();
72
73    let logs = serde_json::from_slice::<Vec<_>>(&payload)
74        .map_err(|_| Error::Invalid(DiscardReason::InvalidJson))?
75        .into_iter()
76        .filter_map(|DeserializableAnnotated(integrity)| {
77            integrity::create_log(integrity, received_at)
78        })
79        .map(|log| {
80            // INGEST-1333: This is only used to keep records in case serializing
81            // the logs back into an envelope fails. Consider removing it.
82            let byte_size = relay_ourlogs::calculate_size(&log);
83
84            records.modify_by(DataCategory::LogItem, 1);
85            records.modify_by(DataCategory::LogByte, byte_size as isize);
86
87            WithHeader::just(log.into())
88        })
89        .collect();
90
91    // Undo all the base item quantities, as they will be completely taken over by the parsed
92    // contents, which contains an arbitrary amount of items (even 0).
93    for (category, quantity) in item.quantities() {
94        records.modify_by(category, -(quantity as isize));
95    }
96
97    let settings = IngestSettings {
98        infer_user_agent: Some(InferUserAgent::Auto),
99        infer_ip: Some(InferIp::Never),
100    };
101
102    create_intermediates(meta, logs, settings)
103}
