17/// The maximum amount of messages accepted in a single Logplex payload.
18///
19/// Logplex sends batches of around 100KB with at most 1024 buffered messages, payloads with
20/// more messages are not sent by Logplex and are rejected as a whole.
21const MAX_MESSAGES: u32 = 10_000;
22
23/// Maximum size/length of the drain token which is added to all logs.
24const MAX_DRAIN_TOKEN: usize = 256;
25
26/// A processor for Heroku log integrations.
27///
28/// Converts Logplex integration items into intermediate log envelopes for further processing.
29#[derive(Debug)]
30pub struct HerokuLogsIntegrationProcessor;
31
32impl processing::Processor for HerokuLogsIntegrationProcessor {
33    type Input = SerializedLogsIntegration;
34    type Output = Nothing;
35    type Error = Error;
36
37    fn cogs() -> relay_cogs::FeatureWeights {
38        AppFeature::Logs.into()
39    }
40
41    fn prepare_envelope(&self, envelope: &mut ManagedEnvelope) -> Option<Managed<Self::Input>> {
42        let item = envelope.envelope_mut().take_item_by(|item| {
43            matches!(
44                item.integration(),
45                Some(Integration::Logs(LogsIntegration::HerokuLogDrain))
46            )
47        })?;
48        let headers = envelope.envelope().headers().clone();
49
50        let work = SerializedLogsIntegration { headers, item };
51        Some(Managed::with_meta_from_managed_envelope(envelope, work))
52    }
53
54    async fn process(
55        &self,
56        work: Managed<Self::Input>,
57        _ctx: processing::Context<'_>,
58    ) -> Result<Output<Self::Output>, Rejected<Self::Error>> {
59        let intermediates = work.try_map(expand)?;
60        Ok(Output::intermediates(intermediates))
61    }
62}
63
64/// Expands a Heroku log integration into an envelope of logs that can be
65/// returned to the processing pipeline.
66pub fn expand(
67    SerializedLogsIntegration { headers, item }: SerializedLogsIntegration,
68    records: &mut RecordKeeper,
69) -> Result<Intermediates, Error> {
70    records.lenient(DataCategory::LogByte);
71
72    let payload = item.payload();
73
74    let msg_count = headers
75        .get_header(HEROKU_LOGPLEX_HEADER_MSG_COUNT)
76        .and_then(|h| h.as_str())
77        .map(|h| h.parse::<u32>())
78        .transpose()
79        .map_err(|_| Error::Invalid(DiscardReason::InvalidLog))?;
80    let drain_token = headers
81        .get_header(HEROKU_LOGPLEX_HEADER_DRAIN_TOKEN)
82        .and_then(|h| h.as_str())
83        .filter(|h| h.len() < MAX_DRAIN_TOKEN);
84
85    let mut count: u32 = 0;
86    let mut logs = Vec::new();
87
88    for frame in relay_ourlogs::parse_logplex_frames(&payload) {
89        count += 1;
90
91        if count > MAX_MESSAGES {
92            return Err(Error::Invalid(DiscardReason::ItemTooLarge(
93                DiscardItemType::Log,
94            )));
95        }
96
97        let frame = frame
98            .inspect_err(|err| {
99                relay_log::debug!(
100                    error = err as &dyn std::error::Error,
101                    "failed to parse Logplex frame"
102                )
103            })
104            .map_err(|_| Error::Invalid(DiscardReason::InvalidLog))?;
105
106        let frame = String::from_utf8_lossy(frame);
107        let message = relay_ourlogs::parse_logplex_message(&frame)
108            .inspect_err(|err| {
109                relay_log::debug!(
110                    error = err as &dyn std::error::Error,
111                    "Failed to parse Logplex message"
112                )
113            })
114            .map_err(|_| Error::Invalid(DiscardReason::InvalidLog))?;
115
116        let log = relay_ourlogs::logplex_message_to_sentry_log(message, drain_token);
117        let byte_size = relay_ourlogs::calculate_size(&log);
118
119        records.modify_by(DataCategory::LogItem, 1);
120        records.modify_by(DataCategory::LogByte, byte_size as isize);
121
122        logs.push(WithHeader::just(log.into()));
123    }
124
125    if msg_count.is_some_and(|expected| expected != count) {
126        relay_log::debug!("Parsed message count differs from Logplex-Msg-Count");
127        return Err(Error::Invalid(DiscardReason::InvalidLog));
128    }
129
130    // Undo all the base item quantities, as they will be completely taken over by the parsed
131    // contents, which contains an arbitrary amount of items (even 0).
132    records.remove(&item);
133
134    let settings = IngestSettings {
135        infer_user_agent: Some(InferUserAgent::Never),
136        infer_ip: Some(InferIp::Never),
137    };
138
139    create_intermediates(headers, logs, settings)
140}
