32impl SentryError for Nswitch {
33    fn event_category(&self) -> DataCategory {
34        DataCategory::Error

37    fn try_expand(items: &mut Vec<Item>, ctx: Context<'_>) -> Result<Option<Expansion<Self>>> {
38        let Some(dying_message) = utils::take_item_by(items, |item| {
39            item.attachment_type() == Some(AttachmentType::NintendoSwitchDyingMessage)
40        }) else {
41            return Ok(None);
42        };
43
44        let mut metrics = Default::default();
45
46        let attachments: Vec<_> = utils::take_items_of_type(items, ItemType::Attachment);
47        let user_reports = utils::take_items_of_type(items, ItemType::UserReport);
48
49        if !ctx.processing.is_processing() {
50            return Ok(Some(Expansion {
51                event: Box::new(utils::take_parsed_event(items, &mut metrics, ctx)?),
52                attachments,
53                user_reports,
54                error: Self::Forward { dying_message },
55                metrics,
56                fully_normalized: false,
57            }));
58        }
59
60        let event = utils::take_item_of_type(items, ItemType::Event);
61
62        let dying_message = expand_dying_message(dying_message.payload())
63            .map_err(ProcessingError::InvalidNintendoDyingMessage)?;
64
65        let mut attachments = attachments;
66        attachments.extend(dying_message.attachments);
67
68        #[cfg_attr(not(feature = "processing"), expect(unused_mut))]
69        let mut event = match (event, dying_message.event) {
70            (Some(event), Some(dying_message)) => {
71                metrics.bytes_ingested_event =
72                    Annotated::new((event.len() + dying_message.len()) as u64);
73                merge_events(event, dying_message, ctx)?
74            }
75            (Some(event), None) => utils::event_from_json_payload(event, None, &mut metrics, ctx)?,
76            (None, Some(event)) => utils::event_from_json_payload(event, None, &mut metrics, ctx)?,
77            (None, None) => return Err(ProcessingError::NoEventPayload.into()),
78        };
79
80        // Reshape switch crash title to match other native platforms, and mark it as fatal.
81        utils::if_processing!(ctx, {
82            use relay_dynamic_config::Feature;
83
84            if let Some(event) = event.value_mut()
85                && ctx
86                    .processing
87                    .project_info
88                    .has_feature(Feature::NintendoEventRewrite)
89            {
90                crate::utils::reshape_switch_crash(event);
91            }
92        });
93
94        Ok(Some(Expansion {
95            event: Box::new(event),
96            attachments,
97            user_reports,
98            error: Self::Process {},
99            metrics,
100            fully_normalized: false,
101        }))
102    }

135/// An error returned when parsing the dying message attachment.
136#[derive(Debug, thiserror::Error)]
137pub enum SwitchProcessingError {
138    #[error("envelope parsing failed")]
139    EnvelopeParsing(#[from] EnvelopeError),
140    #[error("unexpected EOF, expected {expected:?}")]
141    UnexpectedEof { expected: &'static str },
142    #[error("invalid magic number")]
143    InvalidMagic,
144    #[error("invalid {0:?} ({1:?})")]
145    InvalidValue(&'static str, usize),
146    #[error("Zstandard error")]
147    Zstandard(#[source] std::io::Error),
148}

150fn merge_events(
151    from_envelope: Item,
152    from_dying_messages: Item,
153    ctx: Context<'_>,
154) -> Result<Annotated<Event>> {
155    if from_envelope.len().max(from_dying_messages.len()) > ctx.processing.config.max_event_size() {
156        return Err(ProcessingError::PayloadTooLarge(DiscardItemType::Event).into());
157    }
158
159    merge_events_inner(from_envelope, from_dying_messages)
160        .map_err(ProcessingError::InvalidJson)
161        .map_err(Into::into)
162}
163
164fn merge_events_inner(
165    from_envelope: Item,
166    from_dying_messages: Item,
167) -> Result<Annotated<Event>, serde_json::Error> {
168    let from_envelope = serde_json::from_slice(&from_envelope.payload())?;
169    let mut from_dying_message = serde_json::from_slice(&from_dying_messages.payload())?;
170
171    // Uses the dying message as a base and fills it with values from the event.
172    crate::utils::merge_values(&mut from_dying_message, from_envelope);
173
174    Annotated::<Event>::deserialize_with_meta(from_dying_message)
175}

183/// Parses DyingMessage contents and updates the envelope.
184/// See dying_message.md for the documentation.
185fn expand_dying_message(mut payload: Bytes) -> Result<ExpandedDyingMessage, SwitchProcessingError> {
186    // `Item::attachment_type` may report `NintendoSwitchDyingMessage` from an explicitly set item
187    // header, which bypasses the `starts_with(magic)` guard it applies when inferring the type.
188    // Validate the magic here so a crafted short payload can't panic the `advance` below.
189    if !payload.starts_with(NNSWITCH_SENTRY_MAGIC) {
190        return Err(SwitchProcessingError::InvalidMagic);
191    }
192    payload.advance(NNSWITCH_SENTRY_MAGIC.len());
193    let version = payload
194        .try_get_u8()
195        .map_err(|_| SwitchProcessingError::UnexpectedEof {
196            expected: "version",
197        })?;
198
199    match version {
200        0 => expand_dying_message_v0(payload),
201        _ => Err(SwitchProcessingError::InvalidValue(
202            "version",
203            version as usize,
204        )),
205    }
206}
207
208/// DyingMessage protocol v0 parser.
209fn expand_dying_message_v0(
210    mut payload: Bytes,
211) -> Result<ExpandedDyingMessage, SwitchProcessingError> {
212    let encoding_byte = payload
213        .try_get_u8()
214        .map_err(|_| SwitchProcessingError::UnexpectedEof {
215            expected: "encoding",
216        })?;
217    let format = (encoding_byte >> 6) & 0b0000_0011;
218    let compression = (encoding_byte >> 4) & 0b0000_0011;
219    let compression_arg = encoding_byte & 0b0000_1111;
220
221    let compressed_length =
222        payload
223            .try_get_u16()
224            .map_err(|_| SwitchProcessingError::UnexpectedEof {
225                expected: "compressed data length",
226            })?;
227    let data = decompress_data(
228        payload,
229        compressed_length as usize,
230        compression,
231        compression_arg,
232    )?;
233
234    match format {
235        0 => expand_dying_message_from_envelope_items(data),
236        _ => Err(SwitchProcessingError::InvalidValue(
237            "payload format",
238            format as usize,
239        )),
240    }
241}

243/// Merges envelope items with the ones contained in the DyingMessage
244fn expand_dying_message_from_envelope_items(
245    data: Bytes,
246) -> Result<ExpandedDyingMessage, SwitchProcessingError> {
247    let mut items = Envelope::parse_items_bytes(data)
248        .map_err(SwitchProcessingError::EnvelopeParsing)?
249        .into_vec();
250
251    let event = utils::take_item_of_type(&mut items, ItemType::Event);
252    let attachments = utils::take_items_of_type(&mut items, ItemType::Attachment);
253
254    if !items.is_empty() {
255        // Ignore unsupported items instead of failing to keep forward compatibility.
256        relay_log::debug!(
257            "Ignoring {} unsupported items in the dying message",
258            items.len()
259        );
260    }
261
262    Ok(ExpandedDyingMessage { event, attachments })
263}

265fn decompress_data(
266    payload: Bytes,
267    compressed_length: usize,
268    compression: u8,
269    compression_arg: u8,
270) -> Result<Bytes, SwitchProcessingError> {
271    if payload.len() < compressed_length {
272        return Err(SwitchProcessingError::InvalidValue(
273            "compressed data length",
274            compressed_length,
275        ));
276    }
277
278    let data = payload.slice(0..compressed_length);
279    match compression {
280        // No compression
281        0 => Ok(data),
282        // Zstandard
283        1 => decompress_data_zstd(data, compression_arg)
284            .map(Bytes::from)
285            .map_err(SwitchProcessingError::Zstandard),
286        _ => Err(SwitchProcessingError::InvalidValue(
287            "compression format",
288            compression as usize,
289        )),
290    }
291}

293fn get_zstd_dictionary(id: usize) -> Option<&'static zstd::dict::DecoderDictionary<'static>> {
294    // Inlined dictionary binary data.
295    static ZSTD_DICTIONARIES: &[&[u8]] = &[
296        // index 0 = empty dictionary (a.k.a "none")
297        b"",
298    ];
299
300    // We initialize dictionaries (from their binary representation) only once and reuse them when decompressing.
301    static ZSTD_DEC_DICTIONARIES: OnceLock<
302        [zstd::dict::DecoderDictionary; ZSTD_DICTIONARIES.len()],
303    > = OnceLock::new();
304    let dictionaries = ZSTD_DEC_DICTIONARIES.get_or_init(|| {
305        let mut dictionaries: [zstd::dict::DecoderDictionary; ZSTD_DICTIONARIES.len()] =
306            [zstd::dict::DecoderDictionary::new(ZSTD_DICTIONARIES[0])];
307        for i in 0..ZSTD_DICTIONARIES.len() {
308            dictionaries[i] = zstd::dict::DecoderDictionary::new(ZSTD_DICTIONARIES[i]);
309        }
310        dictionaries
311    });
312
313    dictionaries.get(id)
314}
315
316fn decompress_data_zstd(data: Bytes, dictionary_id: u8) -> std::io::Result<Vec<u8>> {
317    let dictionary = get_zstd_dictionary(dictionary_id as usize)
318        .ok_or(std::io::Error::other("Unknown compression dictionary"))?;
319
320    let mut decompressor = ZstdDecompressor::with_prepared_dictionary(dictionary)?;
321    decompressor.decompress(data.as_ref(), MAX_DECOMPRESSED_SIZE)
322}

374        // Note: the attachment length specified in the "outer" envelope attachment is very important.
375        //       Otherwise parsing would fail because the inner one can contain line-breaks.
376        let envelope =
377        r#"{"event_id":"9ec79c33ec9942ab8353589fcb2e04dc","dsn":"https://e12d836b15bb49d7bbf99e64295d995b:@sentry.io/42"}
378{"type":"event"}

383        let mut envelope =
384            Envelope::parse_bytes([Bytes::from(envelope), dying_message].concat().into()).unwrap();
385
386        envelope.take_items_by(|_| true).into_vec()
387    }

457    fn create_compressed_dying_message(encoding: u8) -> Vec<u8> {
458        // The attachment content is as follows:
459        // - 4 bytes magic = sntr
460        // - 1 byte version = 0
461        // - 1 byte encoding
462        // - 2 bytes data length = N bytes - in big endian representation
463        // - N bytes of compressed content (Zstandard)
464        let mut compressor = ZstdCompressor::new(3).unwrap();
465        let compressed_data = compressor
466            .compress(
467                b"\
468                {\"type\":\"event\"}\n\
469                {\"foo\":\"bar\",\"level\":\"info\",\"map\":{\"b\":\"c\"}}\n\
470                {\"type\":\"attachment\",\"length\":2}\n\
471                Hi\n\
472                ",
473            )
474            .unwrap();
475        let mut dying_message: Vec<u8> = Vec::new();
476        dying_message.write_all(b"sntr\0").unwrap();
477        dying_message.write_all(&[encoding]).unwrap();
478        dying_message
479            .write_all(&(compressed_data.len() as u16).to_be_bytes())
480            .unwrap();
481        dying_message.write_all(&compressed_data).unwrap();
482        dying_message
483    }

529    #[test]
530    fn test_switch_crash_is_reshaped() {
531        let mut event = Item::new(ItemType::Event);
532        event.set_payload(
533            ContentType::Json,
534            include_bytes!(
535                "../../../../../tests/integration/fixtures/native/nnswitch_crportal_event.json"
536            )
537            .as_slice(),
538        );
539
540        let mut dying_message = Item::new(ItemType::Attachment);
541        dying_message.set_filename(NNSWITCH_DYING_MESSAGE_FILENAME);
542        dying_message.set_attachment_type(AttachmentType::NintendoSwitchDyingMessage);
543        dying_message.set_payload(
544            ContentType::OctetStream,
545            include_bytes!(
546                "../../../../../tests/integration/fixtures/native/nnswitch_crportal_dying_message_raw.dat"
547            )
548            .as_slice(),
549        );
550
551        let mut items = vec![event, dying_message];
552
553        let parsed = Nswitch::try_expand(&mut items, ctx()).unwrap().unwrap();
554
555        let event = parsed.event.value().unwrap();
556
557        // The DyingMessage scope patch is the merge base and contributes the fields the
558        // forwarded event lacks.
559        assert_eq!(
560            event.release.value().map(|release| release.as_str()),
561            Some("test-app@1.0.0")
562        );
563        assert_eq!(
564            event.environment.value().map(String::as_str),
565            Some("integration-test")
566        );
567
568        // The crash is rendered as fatal: both the forwarded event and the scope patch carry
569        // `level: error`, and the reshape asserts the severity of a captured crash.
570        assert_eq!(event.level.value(), Some(&Level::Fatal));
571
572        let exception = event
573            .exceptions
574            .value()
575            .unwrap()
576            .values
577            .value()
578            .unwrap()
579            .last()
580            .unwrap()
581            .value()
582            .unwrap();
583
584        // Marked synthetic, so Sentry drops the result-code `type` from the title and falls
585        // back to the crashing function. Nintendo's own `handled: false` passes through.
586        let mechanism = exception.mechanism.value().unwrap();
587        assert_eq!(mechanism.synthetic.value(), Some(&true));
588        assert_eq!(mechanism.handled.value(), Some(&false));
589
590        // The result code and its description are preserved; only the `type`'s influence on the
591        // title is removed. `value` remains as the issue subtitle.
592        assert_eq!(
593            exception.ty.value().map(String::as_str),
594            Some("2168-0002 ResultAccessViolationData")
595        );
596    }
