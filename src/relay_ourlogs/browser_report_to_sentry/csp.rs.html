11fn scheme_uses_host(scheme: &str) -> bool {
12    // List of schemes with host (netloc) from Python's urlunsplit:
13    // see <https://github.com/python/cpython/blob/1eac437e8da106a626efffe9fce1cb47dbf5be35/Lib/urllib/parse.py#L51>
14    //
15    // Only modification: "" is set to false, since there is a separate check in the urlunsplit
16    // implementation that omits the leading "//" in that case.
17    matches!(
18        scheme,
19        "ftp"
20            | "http"
21            | "gopher"
22            | "nntp"
23            | "telnet"
24            | "imap"
25            | "wais"
26            | "file"
27            | "mms"
28            | "https"
29            | "shttp"
30            | "snews"
31            | "prospero"
32            | "rtsp"
33            | "rtspu"
34            | "rsync"
35            | "svn"
36            | "svn+ssh"
37            | "sftp"
38            | "nfs"
39            | "git"
40            | "git+ssh"
41            | "ws"
42            | "wss"
43    )
44}

46/// Mimicks Python's urlunsplit with all its quirks.
47fn unsplit_url(scheme: &str, host: &str) -> String {
48    if !host.is_empty() || scheme_uses_host(scheme) {
49        format!("{scheme}://{host}")
50    } else if !scheme.is_empty() {
51        format!("{scheme}:{host}")
52    } else {
53        String::new()
54    }
55}

57fn normalize_url(value: &str) -> Cow<'_, str> {
58    // A lot of these values get reported as literally just the scheme (see
59    // <https://w3c.github.io/webappsec-csp/#strip-url-for-use-in-reports>). So a value like 'data'
60    // or 'blob', which are valid schemes, just not a uri. So we want to normalize it into a
61    // URL.
62
63    if !value.contains(':') {
64        return Cow::Owned(unsplit_url(value, ""));
65    }
66
67    let url = match Url::parse(value) {
68        Ok(url) => url,
69        Err(_) => return Cow::Borrowed(value),
70    };
71
72    let normalized = match url.scheme() {
73        "http" | "https" => Cow::Borrowed(url.host_str().unwrap_or_default()),
74        scheme => Cow::Owned(unsplit_url(scheme, url.host_str().unwrap_or_default())),
75    };
76
77    Cow::Owned(match url.port() {
78        Some(port) => format!("{normalized}:{port}"),
79        None => normalized.into_owned(),
80    })
81}

83fn get_message(blocked_url: &BlockedUrl, directive: &CspDirective) -> String {
84    match blocked_url {
85        BlockedUrl::Inline => format!("Blocked unsafe inline '{directive}'"),
86        BlockedUrl::Eval => format!("Blocked unsafe eval() '{directive}'"),
87        BlockedUrl::WasmEval => format!("Blocked unsafe WASM eval() '{directive}'"),
88        BlockedUrl::TrustedTypesPolicy => "Blocked creation of trusted types policy".to_owned(),
89        BlockedUrl::TrustedTypesSink => "Blocked sending of unsanitized data to sink".to_owned(),
90        BlockedUrl::Url(url) => {
91            let url = normalize_url(url);
92
93            match directive {
94                CspDirective::ChildSrc => format!("Blocked 'child' from '{url}'"),
95                CspDirective::ConnectSrc => format!("Blocked 'connect' from '{url}'"),
96                CspDirective::FontSrc => format!("Blocked 'font' from '{url}'"),
97                CspDirective::FormAction => format!("Blocked 'form' action to '{url}'"),
98                CspDirective::ImgSrc => format!("Blocked 'image' from '{url}'"),
99                CspDirective::ManifestSrc => format!("Blocked 'manifest' from '{url}'"),
100                CspDirective::MediaSrc => format!("Blocked 'media' from '{url}'"),
101                CspDirective::ObjectSrc => format!("Blocked 'object' from '{url}'"),
102                CspDirective::ScriptSrc => format!("Blocked 'script' from '{url}'"),
103                CspDirective::ScriptSrcAttr => {
104                    format!("Blocked inline script attribute from '{url}'")
105                }
106                CspDirective::ScriptSrcElem => format!("Blocked 'script' from '{url}'"),
107                CspDirective::StyleSrc => format!("Blocked 'style' from '{url}'"),
108                CspDirective::StyleSrcElem => format!("Blocked 'style' from '{url}'"),
109                CspDirective::StyleSrcAttr => format!("Blocked style attribute from '{url}'"),
110                CspDirective::Other(_) => format!("Blocked '{directive}' from '{url}'"),
111            }
112        }
113    }
114}

116pub fn create_log_from_csp_body(body: Body, mut attributes: Attributes) -> OurLog {
117    let Body {
118        document_url,
119        referrer,
120        blocked_url,
121        effective_directive,
122        original_policy,
123        source_file,
124        sample,
125        disposition,
126        status_code,
127    } = body;
128
129    add_string_attribute!(attributes, BROWSER__REPORT__BLOCKED_URL, blocked_url);
130    add_string_attribute!(attributes, BROWSER__REPORT__DOCUMENT_URL, document_url);
131    add_attribute!(
132        attributes,
133        BROWSER__REPORT__REPORT_ONLY,
134        matches!(disposition, Disposition::Report)
135    );
136    add_string_attribute!(
137        attributes,
138        BROWSER__REPORT__CSP__ORIGINAL_POLICY,
139        original_policy
140    );
141    add_string_attribute!(
142        attributes,
143        BROWSER__REPORT__CSP__EFFECTIVE_DIRECTIVE,
144        effective_directive
145    );
146
147    if let Some(sample) = sample {
148        add_string_attribute!(attributes, BROWSER__REPORT__CSP__SAMPLE, sample);
149    }
150
151    if let Some(SourceFile {
152        source_file,
153        column_number,
154        line_number,
155    }) = source_file
156    {
157        add_string_attribute!(attributes, CODE__FILE__PATH, source_file);
158        if let Some(column_number) = column_number {
159            add_attribute!(attributes, CODE__COLUMN__NUMBER, column_number as i64);
160        }
161        if let Some(line_number) = line_number {
162            add_attribute!(attributes, CODE__LINE__NUMBER, line_number as i64);
163        }
164    }
165
166    add_attribute!(attributes, HTTP__RESPONSE__STATUS_CODE, status_code as i64);
167
168    if let Some(referrer) = referrer
169        && !referrer.is_empty()
170    {
171        let referrer_key = interpolate::http__request__header__key("referer");
172        add_attribute!(attributes, referrer_key, vec![referrer.clone()]);
173    }
174
175    OurLog {
176        body: get_message(&blocked_url, &effective_directive).into(),
177        level: OurLogLevel::Warn.into(),
178        attributes: attributes.into(),
179        ..Default::default()
180    }
181}
