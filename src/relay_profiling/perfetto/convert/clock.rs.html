8/// Tracks the clocks of a Perfetto trace.
9///
10/// Like the Perfetto trace processor, timestamps are converted with the clock snapshots seen
11/// up until the packet. The offsets between clocks change over time, for example `MONOTONIC`
12/// does not advance while the device is suspended, which is why only the latest snapshot is
13/// used.
14///
15/// See <https://perfetto.dev/docs/concepts/clock-sync>.
16#[derive(Debug, Default)]
17pub struct Clocks {
18    /// Clock defaulted per sequence via `TracePacketDefaults`.
19    defaults: HashMap<SequenceId, ClockId>,
20    /// The primary trace clock, as declared by the latest snapshot declaring one.
21    primary: Option<ClockId>,
22    /// The latest clock snapshot of the trace which can be converted to real time.
23    snapshot: Option<Snapshot>,
24}
25
26impl Clocks {
27    /// Records the sequence defaults contained in the `packet`.
28    ///
29    /// Like interned data, sequence defaults are part of the incremental state of a sequence and
30    /// are dropped when it is cleared.
31    pub fn update_defaults(&mut self, packet: &proto::TracePacket) -> Result<(), BudgetExceeded> {
32        let seq_id = SequenceId::new(packet);
33
34        // Technically this is off by-one, but significantly easier to handle.
35        if self.defaults.len() >= consts::MAX_SEQUENCE_IDS {
36            return Err(BudgetExceeded);
37        }
38
39        if utils::has_incremental_state_cleared(packet) {
40            self.defaults.remove(&seq_id);
41        }
42
43        let Some(defaults) = packet.trace_packet_defaults else {
44            return Ok(());
45        };
46        match defaults.timestamp_clock_id {
47            Some(clock_id) => self.defaults.insert(seq_id, ClockId::new(clock_id)),
48            None => self.defaults.remove(&seq_id),
49        };
50
51        Ok(())
52    }
53
54    /// Records a [`proto::ClockSnapshot`], replacing the previously recorded snapshot.
55    ///
56    /// Snapshots without `REALTIME` cannot be converted to real time and are ignored entirely.
57    pub fn update_snapshot(&mut self, cs: &proto::ClockSnapshot) {
58        let snapshot = Snapshot {
59            // Builtin clocks only, see `ClockId::is_builtin`.
60            timestamps: cs
61                .clocks
62                .iter()
63                .filter_map(|clock| Some((ClockId::new(clock.clock_id?), clock.timestamp?)))
64                .filter(|(clock_id, _)| clock_id.is_builtin())
65                .collect(),
66        };
67
68        if !snapshot.timestamps.contains_key(&ClockId::REALTIME) {
69            return;
70        }
71
72        if let Some(primary) = cs.primary_trace_clock {
73            self.primary = Some(ClockId::new(primary));
74        }
75        self.snapshot = Some(snapshot);
76    }
77
78    /// Converts the timestamp of the `packet` to real time in nanoseconds.
79    ///
80    /// Returns `None` if the clock of the `packet` cannot be converted to real time, for example
81    /// because no clock snapshot was recorded yet.
82    pub fn to_realtime(&self, packet: &proto::TracePacket) -> Option<i128> {
83        let timestamp = i128::from(packet.timestamp.unwrap_or(0));
84
85        let clock_id = self.clock_id(packet);
86        if clock_id == ClockId::REALTIME {
87            return Some(timestamp);
88        }
89
90        let snapshot = self.snapshot.as_ref()?;
91        let realtime = snapshot.timestamps.get(&ClockId::REALTIME).copied()?;
92        let clock = snapshot.timestamps.get(&clock_id).copied()?;
93
94        Some(timestamp + i128::from(realtime) - i128::from(clock))
95    }
96
97    /// Returns the clock the timestamp of the `packet` is recorded in.
98    ///
99    /// The clock of a packet is, in order of precedence, its own `timestamp_clock_id`, the
100    /// `timestamp_clock_id` defaulted for its sequence, or the primary trace clock.
101    fn clock_id(&self, packet: &proto::TracePacket) -> ClockId {
102        match packet.timestamp_clock_id {
103            Some(clock_id) => ClockId::new(clock_id),
104            None => self
105                .defaults
106                .get(&SequenceId::new(packet))
107                .copied()
108                .unwrap_or_else(|| self.primary()),
109        }
110    }
111
112    /// Returns the primary trace clock, which defaults to `BOOTTIME`.
113    fn primary(&self) -> ClockId {
114        self.primary.unwrap_or(ClockId::BOOTTIME)
115    }
116}

118/// A Perfetto clock id.
119#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
120pub struct ClockId(u32);
121
122impl ClockId {
123    /// Perfetto builtin real time clock.
124    const REALTIME: Self = Self(consts::CLOCK_REALTIME);
125    /// Perfetto builtin boot time clock.
126    const BOOTTIME: Self = Self(consts::CLOCK_BOOTTIME);
127
128    /// Creates a new [`Self`] from a raw Perfetto clock id.
129    fn new(clock_id: u32) -> Self {
130        Self(clock_id)
131    }
132
133    /// Returns `true` if this is one of the globally known, builtin clocks.
134    ///
135    /// Clocks beyond the builtin range are custom clocks, either scoped to a single sequence or
136    /// global. They may carry a unit multiplier or be incremental, neither of which is decoded
137    /// here.
138    fn is_builtin(self) -> bool {
139        self.0 <= consts::BUILTIN_CLOCK_MAX_ID
140    }
141}

143/// The clocks of a trace at a single point in time.

147    timestamps: HashMap<ClockId, u64>,
