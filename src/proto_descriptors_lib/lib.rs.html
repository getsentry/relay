9// Runs the rendering process, given the supplied descriptor set, and a set of root protos from
10// which to start the traversal.  Returns the String of the rendered file, or an error.
11pub fn render(
12    root_files: &[String],
13    use_stmts: &Option<String>,
14    descriptor_set_bytes: &[u8],
15) -> Result<String, anyhow::Error> {
16    let descriptor_set =
17        FileDescriptorSet::decode(descriptor_set_bytes).context("malformed descriptor set")?;
18
19    let (root_messages, messages) = extract_roots_and_messages(root_files, &descriptor_set)?;
20    let reachable = walk(messages)?;
21    let rendered = render_to_rust(use_stmts, &reachable, &root_messages)?;
22
23    Ok(rendered)
24}

26// Roots are the top-level messages in the specified file, but we'll still transitively pull
27// in other messages that are nested.  Ultimately, we only want to implement Decodable for root
28// types (rather than everything transitive, which is a lot.)
29fn extract_roots_and_messages<'a>(
30    root_files: &[String],
31    descriptor_set: &'a FileDescriptorSet,
32) -> Result<(Vec<String>, BTreeMap<String, &'a DescriptorProto>)> {
33    let mut messages = BTreeMap::new();
34    let mut root_types = vec![];
35    for file in &descriptor_set.file {
36        let prefix = match file.package() {
37            "" => String::new(),
38            package => format!(".{package}"),
39        };
40
41        for message in &file.message_type {
42            collect(&prefix, message, &mut messages)?;
43
44            if root_files.contains(&file.name().to_owned()) {
45                root_types.push(message.name().to_owned());
46            }
47        }
48    }
49
50    Ok((root_types, messages))
51}

53// Adds `message` and everything transitively nested inside it to `messages`.
54fn collect<'a>(
55    prefix: &str,
56    message: &'a DescriptorProto,
57    messages: &mut BTreeMap<String, &'a DescriptorProto>,
58) -> Result<()> {
59    let full_name = format!("{prefix}.{}", message.name());
60
61    if messages.insert(full_name.clone(), message).is_some() {
62        bail!("{full_name} is defined twice in the descriptor set");
63    }
64
65    for nested in &message.nested_type {
66        collect(&full_name, nested, messages)?;
67    }
68
69    Ok(())
70}

72struct ProcessedMessage<'a> {
73    // The fully qualified proto name
74    full_name: String,
75    // The short name of this message
76    ident: String,
77    // `(tag, fully qualified name)` for every field holding a nested message, ordered by tag.
78    nested: Vec<(u32, &'a str)>,
79}
80
81fn walk<'a>(messages: BTreeMap<String, &'a DescriptorProto>) -> Result<Vec<ProcessedMessage<'a>>> {
82    let mut processed = Vec::new();
83    let mut queue: Vec<String> = messages.keys().map(|k| k.to_owned()).collect();
84
85    while let Some(full_name) = queue.pop() {
86        let (full_name, message) = messages
87            .get_key_value(&full_name)
88            .map(|(name, message)| (name.as_str(), *message))
89            .with_context(|| format!("{full_name} is not in the descriptor set"))?;
90
91        let mut nested = Vec::new();
92        for field in &message.field {
93            // Bail loudly on this for now; not supported in proto3.
94            if field.r#type() == Type::Group {
95                bail!(
96                    "{full_name}.{} is a group, which this generator does not describe",
97                    field.name()
98                );
99            }
100
101            if field.r#type() != Type::Message {
102                continue;
103            }
104
105            let type_name = field.type_name();
106
107            let target = messages.get(type_name).with_context(|| {
108                format!(
109                    "{full_name}.{} refers to {type_name}, which is not in the descriptor set - \
110                     was it built without --include_imports?",
111                    field.name()
112                )
113            })?;
114
115            // Maps not supported at the moment.
116            if target.options.as_ref().is_some_and(|o| o.map_entry()) {
117                bail!(
118                    "{full_name}.{} is a map, which this generator does not describe",
119                    field.name()
120                );
121            }
122
123            let tag = u32::try_from(field.number())
124                .with_context(|| format!("{full_name}.{} has a negative tag", field.name()))?;
125
126            nested.push((tag, type_name));
127        }
128
129        nested.sort_unstable();
130
131        processed.push(ProcessedMessage {
132            full_name: full_name.trim_start_matches('.').to_owned(),
133            ident: ident(full_name),
134            nested,
135        });
136    }
137
138    // Sorting by name keeps regeneration diffs empty when nothing has actually changed.
139    processed.sort_unstable_by(|a, b| a.full_name.cmp(&b.full_name));
140
141    let mut idents = BTreeMap::new();
142    for message in &processed {
143        if let Some(other) = idents.insert(message.ident.clone(), message.full_name.clone()) {
144            bail!(
145                "{} and {} both map to the static {}",
146                other,
147                message.full_name,
148                message.ident
149            );
150        }
151    }
152
153    Ok(processed)
154}

156// Derives the name of a static from a fully qualified proto name. For example,
157// `.opentelemetry.proto.trace.v1.Span.Event` becomes `SPAN_EVENT`.
158fn ident(full_name: &str) -> String {
159    // Message names start with an uppercase letter and package segments do not, so the first
160    // uppercase segment is where the message path begins.
161    full_name
162        .split('.')
163        .skip_while(|segment| !segment.starts_with(|c: char| c.is_ascii_uppercase()))
164        .map(|s| s.to_case(Case::UpperSnake))
165        .collect::<Vec<_>>()
166        .join("_")
167}

169// The actual rendering function producing the generated file.
170fn render_to_rust(
171    use_stmts: &Option<String>,
172    reachable: &[ProcessedMessage<'_>],
173    root_types: &Vec<String>,
174) -> Result<String> {
175    let mut out = String::new();
176
177    writeln!(
178        out,
179        "//! @generated by tools/proto-descriptors - DO NOT EDIT."
180    )?;
181
182    writeln!(out)?;
183    writeln!(out, "use relay_serialization::prost::Error;")?;
184    writeln!(out, "use relay_serialization::prost::MessageDesc;")?;
185    writeln!(out, "use relay_serialization::prost::decode;")?;
186
187    if let Some(stmts) = use_stmts {
188        writeln!(out, "{}", stmts)?;
189    }
190
191    writeln!(
192        out,
193        "/// Provides a bounded decode function for any implementor."
194    )?;
195    writeln!(out, "pub trait Decodable {{")?;
196    writeln!(
197        out,
198        "/// Attempts to decode the provided proto bytes within the specified number of ops."
199    )?;
200    writeln!(
201        out,
202        "fn decode_bounded(buf: &[u8], max_ops: usize) -> Result<Self, Error> where Self: Sized; }}"
203    )?;
204
205    for message in reachable {
206        let nested = if !message.nested.is_empty() {
207            let mut nested = "".to_owned();
208            for (tag, type_name) in &message.nested {
209                nested += &format!("        ({tag}, &{}),", ident(type_name));
210            }
211            nested
212        } else {
213            "".to_owned()
214        };
215
216        writeln!(out)?;
217        writeln!(out, "/// `{}`", message.full_name)?;
218        writeln!(
219            out,
220            "pub static {}: MessageDesc = MessageDesc {{",
221            message.ident
222        )?;
223        writeln!(out, "    name: \"{}\",", message.full_name)?;
224        writeln!(out, "    nested: &[{}],", nested)?;
225        writeln!(out, "}};")?;
226    }
227
228    // We only implement Deocodable for root_types.
229    for typ in root_types {
230        writeln!(
231            out,
232            "
233            impl Decodable for {} {{\
234                fn decode_bounded(buf: &[u8], max_ops: usize) -> Result<Self, Error> {{ decode(buf, &{}, max_ops) }}\
235            }}",
236            typ,
237            typ.to_case(Case::UpperSnake)
238        )?;
239    }
240
241    Ok(out)
242}

259    /// Compiles `protos` into a descriptor set, the way the build scripts that feed [`render`]
260    /// in production get theirs. Imports are resolved transitively, so the result matches what
261    /// `protoc --include_imports` produces.
262    fn file_descriptor_set(protos: &[&str]) -> FileDescriptorSet {
263        protox::compile(protos, [proto_dir()]).unwrap()
264    }
265
266    /// The encoded bytes [`render`] actually takes.
267    fn descriptor_set(protos: &[&str]) -> Vec<u8> {
268        file_descriptor_set(protos).encode_to_vec()
269    }

272    fn render_fixture(protos: &[&str]) -> Result<String> {

274        render(&roots, &None, &descriptor_set(protos))

277    /// Drops all whitespace, so assertions do not depend on the generated file\'s (pre-rustfmt)
278    /// layout.
279    fn squeeze(s: &str) -> String {
280        s.chars().filter(|c| !c.is_whitespace()).collect()
281    }
282
283    /// The `MessageDesc` static [`render_to_rust`] is expected to emit for one message.
284    fn desc(ident: &str, name: &str, nested: &[(u32, &str)]) -> String {
285        let nested: String = nested
286            .iter()
287            .map(|(tag, ident)| format!("({tag}, &{ident}),"))
288            .collect();
289
290        squeeze(&format!(
291            "pub static {ident}: MessageDesc = MessageDesc {{ \
292                 name: \"{name}\", \
293                 nested: &[{nested}], \
294             }};"
295        ))
296    }
297
298    /// The `Decodable` impl [`render_to_rust`] is expected to emit for one root type.
299    fn decodable_impl(typ: &str, ident: &str) -> String {
300        squeeze(&format!(
301            "impl Decodable for {typ} {{ \
302                 fn decode_bounded(buf: &[u8], max_ops: usize) -> Result<Self, Error> {{ \
303                     decode(buf, &{ident}, max_ops) \
304                 }} \
305             }}"
306        ))
307    }

309    /// The idents of the statics, in the order they appear in the generated file.
310    fn static_idents(rendered: &str) -> Vec<&str> {
311        rendered
312            .lines()
313            .filter_map(|line| line.trim().strip_prefix("pub static "))
314            .filter_map(|line| line.split(':').next())
315            .collect()
316    }

333    #[test]
334    fn test_descriptors_describe_every_nested_message() {
335        let rendered = squeeze(&render_fixture(&[TREE]).unwrap());
336
337        // Only message-typed fields are described; `Leaf`\'s strings, scalars, packed
338        // `repeated uint64`, `bytes` and enum all carry their own length or are fixed width, so
339        // they cost a bounded number of ops and do not need a descriptor. Tag 9 is the `Leaf`
340        // arm of the `oneof`, which recurses.
341        assert!(rendered.contains(&desc("LEAF", "relay.test.v1.Leaf", &[(9, "LEAF")])));
342
343        // Repeated and singular message fields are described identically - the bound is per
344        // occurrence - so `root` and `forest` both point at `TREE_NODE`. Tag 3 resolves into the
345        // imported file.
346        assert!(rendered.contains(&desc(
347            "TREE",
348            "relay.test.v1.Tree",
349            &[(1, "TREE_NODE"), (2, "TREE_NODE"), (3, "METADATA")],
350        )));
351
352        // A type nested inside another message is flattened into a top-level static, and
353        // recursion through it is just a self-reference.
354        assert!(rendered.contains(&desc(
355            "TREE_NODE",
356            "relay.test.v1.Tree.Node",
357            &[(2, "LEAF"), (3, "TREE_NODE")],
358        )));
359
360        // Transitively reachable messages from the imported file are described too, even though
361        // that file is not a root.
362        assert!(rendered.contains(&desc(
363            "METADATA",
364            "relay.test.common.v1.Metadata",
365            &[(2, "TAG")],
366        )));
367        assert!(rendered.contains(&desc("TAG", "relay.test.common.v1.Tag", &[])));
368    }
